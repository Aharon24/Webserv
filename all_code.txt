#ifndef HTTP_RESPONSE_HPP
#define HTTP_RESPONSE_HPP

#include <ctime>
#include <map>
#include <string>

/*
** HttpResponse
**
** Owns everything about the bytes that go back to the client:
** status line, headers (Content-Type, Content-Length, Date, ...) and body.
** Other modules (Router, StaticFileHandler, Server, future CGI/Upload code)
** fill an HttpResponse and call build() / build_head() to serialize it.
*/
class HttpResponse
{
private:
    int _status;
    std::string _reason;
    std::map<std::string, std::string> _headers;
    std::string _body;

public:
    HttpResponse();
    explicit HttpResponse(int status);

    // Status
    void set_status(int status);
    void set_status(int status, const std::string &reason);
    int get_status() const;
    const std::string &get_reason() const;

    // Headers / body
    void set_header(const std::string &name, const std::string &value);
    bool has_header(const std::string &name) const;
    std::string get_header(const std::string &name) const;
    void set_body(const std::string &body, const std::string &content_type);
    const std::string &get_body() const;

    // Serialization.
    // build_head(): status line + headers + empty line (no body).
    // build():      build_head() + body (body is omitted for 1xx/204/304).
    std::string build_head(const std::string &version) const;
    std::string build(const std::string &version) const;

    // Static helpers
    static std::string reason_phrase(int status);
    static std::string mime_type(const std::string &path);
    static std::string http_date(time_t when);
    static std::string html_escape(const std::string &text);
    static bool status_has_no_body(int status);

    // Ready-made responses
    static HttpResponse redirect(int status, const std::string &location);
    static HttpResponse default_error(int status);

    // Adapter for handlers that still return a "404 Not Found" style
    // status string and a raw body (POST / DELETE code of other modules).
    static HttpResponse from_status_line(const std::string &status_line, const std::string &body);
};

#ifndef ROUTER_HPP
#define ROUTER_HPP

#include <string>

#include "HttpRequest.hpp"
#include "ServerConfig.hpp"

enum RouteStatus
{
    ROUTE_OK,                   // request may be handled by a method handler
    ROUTE_REDIRECT,             // location has "return": send redirect_code/redirect_url
    ROUTE_BAD_REQUEST,          // malformed percent-encoding or NUL byte in the URI
    ROUTE_FORBIDDEN,            // URI tries to climb above the root ("/../")
    ROUTE_METHOD_NOT_ALLOWED    // method is not in the location's allowed_methods
};

enum NormalizeStatus
{
    NORMALIZE_OK,
    NORMALIZE_BAD_REQUEST,
    NORMALIZE_FORBIDDEN
};

// Everything a method handler needs to know about where a request points.
struct RouteResult
{
    RouteStatus status;
    const LocationConfig *location;     // matched location, NULL if none matched
    std::string uri;                    // decoded + normalized path, always starts with '/'
    std::string root;                   // effective root directory
    std::string fs_path;                // filesystem path built from root + uri
    std::string allow;                  // "GET, POST" (filled for 405)
    int redirect_code;                  // filled for ROUTE_REDIRECT
    std::string redirect_url;

    RouteResult()
        : status(ROUTE_OK), location(NULL), redirect_code(0)
    {
    }
};

/*
** Router
**
** Decides which location block a request belongs to and where on the disk
** the requested resource lives. It never touches the disk itself.
*/
class Router
{
private:
    const ServerConfig &_conf;

    Router &operator=(const Router &other);

public:
    explicit Router(const ServerConfig &conf);
    Router(const Router &other);
    ~Router();

    // Full routing decision for one request.
    RouteResult route(const HttpRequest &request) const;

    // Longest-prefix match on whole path segments ("/img" never matches "/imgx").
    const LocationConfig *find_location(const std::string &uri) const;

    // Effective settings of a location, falling back to the server block.
    std::string get_root(const LocationConfig *location) const;
    std::string get_index(const LocationConfig *location) const;

    // root + (uri without the location prefix)
    std::string build_path(const std::string &uri, const LocationConfig *location) const;

    // Empty allowed_methods means "everything we implement" (GET, POST, DELETE).
    bool is_method_allowed(const LocationConfig *location, const std::string &method) const;
    std::string allowed_methods_header(const LocationConfig *location) const;

    // Percent-decode once, drop "." / empty segments, resolve "..", keep a trailing '/'.
    static NormalizeStatus normalize_uri(const std::string &raw, std::string &normalized);

    // Percent-encode everything except unreserved characters and '/'.
    static std::string encode_uri(const std::string &path);
};

namespace
{
    void close_fd(int &fd)
    {
        if (fd != -1)
        {
            close(fd);
            fd = -1;
        }
    }

    bool set_nonblocking(int fd)
    {
        int flags = fcntl(fd, F_GETFL, 0);

        if (flags == -1)
            return false;

        return fcntl(fd, F_SETFL, flags | O_NONBLOCK) != -1;
    }
    bool set_cloexec(int fd)
    {
        int flags = fcntl(fd, F_GETFD, 0);

        if (flags == -1)
            return false;

        return fcntl(fd, F_SETFD, flags | FD_CLOEXEC) != -1;
    }

    std::string number_to_string(std::size_t number)
    {
        std::ostringstream ss;
        ss << number;
        return ss.str();
    }

    std::vector<std::string> build_environment(
        const HttpRequest &request,
        const std::string &script_path,
        const std::string &path_info)
    {
        std::vector<std::string> env;

        env.push_back("GATEWAY_INTERFACE=CGI/1.1");
        env.push_back("SERVER_SOFTWARE=webserv/1.0");
        env.push_back("SERVER_PROTOCOL=" + request.version);
        env.push_back("REQUEST_METHOD=" + request.method);
        env.push_back("QUERY_STRING=" + request.query_string);
        env.push_back("SCRIPT_NAME=" + request.path);
        env.push_back("SCRIPT_FILENAME=" + script_path);
        env.push_back("PATH_INFO=" + path_info);
        env.push_back("CONTENT_LENGTH=" + number_to_string(request.body.size()));

        std::map<std::string, std::string>::const_iterator it;

        it = request.headers.find("content-type");

        if (it != request.headers.end())
            env.push_back("CONTENT_TYPE=" + it->second);
        else
            env.push_back("CONTENT_TYPE=");

        env.push_back(
            "REQUEST_URI=" + request.path +
            (request.query_string.empty() ? "" : "?" + request.query_string)
        );

        for (it = request.headers.begin(); it != request.headers.end(); ++it)
        {
            if (it->first == "content-type" ||
                it->first == "content-length")
                continue;

            std::string name = "HTTP_";

            for (std::size_t i = 0; i < it->first.size(); ++i)
            {
                char c = it->first[i];

                if (c == '-')
                    name += '_';
                else if (c >= 'a' && c <= 'z')
                    name += static_cast<char>(c - 'a' + 'A');
                else
                    name += c;
            }

            env.push_back(name + "=" + it->second);
        }

        return env;
    }
}

bool CgiHandler::start(std::string &error_status)
{
    error_status = "500 Internal Server Error";

    if (_pid != -1)
        return false;

    if (_script_path.empty() || _interpreter_path.empty() ||
        _script_path.find('\0') != std::string::npos ||
        _interpreter_path.find('\0') != std::string::npos)
        return false;

    struct stat info;

    if (stat(_script_path.c_str(), &info) == -1)
    {
        error_status = "404 Not Found";
        return false;
    }

    if (!S_ISREG(info.st_mode))
    {
        error_status = "403 Forbidden";
        return false;
    }

    if (access(_script_path.c_str(), R_OK) == -1)
    {
        error_status = "403 Forbidden";
        return false;
    }

    std::string::size_type slash = _script_path.find_last_of('/');

    std::string script_dir = ".";
    std::string script_name = _script_path;

    if (slash != std::string::npos)
    {
        if (slash == 0)
            script_dir = "/";
        else
            script_dir = _script_path.substr(0, slash);

        script_name = _script_path.substr(slash + 1);
    }

    if (script_name.empty())
        return false;

    std::vector<std::string> env =
        build_environment(_request, _script_path, _path_info);

    std::vector<char *> envp;

    for (std::size_t i = 0; i < env.size(); ++i)
        envp.push_back(const_cast<char *>(env[i].c_str()));

    envp.push_back(NULL);

    char *argv[3];

    argv[0] = const_cast<char *>(_interpreter_path.c_str());
    argv[1] = const_cast<char *>(script_name.c_str());
    argv[2] = NULL;

    int input_pipe[2] = {-1, -1};
    int output_pipe[2] = {-1, -1};

    if (pipe(input_pipe) == -1)
    {
        error_status = "500 Internal Server Error";
        return false;
    }

    if (pipe(output_pipe) == -1)
    {
        close_fd(input_pipe[0]);
        close_fd(input_pipe[1]);
        return false;
    }

    for (int i = 0; i < 2; ++i)
    {
        if (!set_cloexec(input_pipe[i]) ||
            !set_cloexec(output_pipe[i]))
        {
            close_fd(input_pipe[0]);
            close_fd(input_pipe[1]);
            close_fd(output_pipe[0]);
            close_fd(output_pipe[1]);
            return false;
        }
    }

    _pid = fork();

    if (_pid == -1)
    {
        close_fd(input_pipe[0]);
        close_fd(input_pipe[1]);
        close_fd(output_pipe[0]);
        close_fd(output_pipe[1]);
        return false;
    }

    if (_pid == 0)
    {
        close(input_pipe[1]);
        close(output_pipe[0]);

        if (dup2(input_pipe[0], STDIN_FILENO) == -1)
            _exit(127);

        if (dup2(output_pipe[1], STDOUT_FILENO) == -1)
            _exit(127);

        if (input_pipe[0] != STDIN_FILENO)
            close(input_pipe[0]);

        if (output_pipe[1] != STDOUT_FILENO)
            close(output_pipe[1]);

        if (_client_fd >= 0)
            close(_client_fd);

        signal(SIGPIPE, SIG_DFL);
        if (chdir(script_dir.c_str()) == -1)
            _exit(127);

        execve(_interpreter_path.c_str(), argv, &envp[0]);

        _exit(127);
    }

    close_fd(input_pipe[0]);
    close_fd(output_pipe[1]);

    _stdin_fd = input_pipe[1];
    _stdout_fd = output_pipe[0];

    if (!set_nonblocking(_stdin_fd) ||
        !set_nonblocking(_stdout_fd))
    {
        kill(_pid, SIGKILL);
        waitpid(_pid, &_wait_status, 0);

        _child_exited = true;
        _pid = -1;

        close_fd(_stdin_fd);
        close_fd(_stdout_fd);

        return false;
    }

    _started_at = time(NULL);

    if (_input_buffer.empty())
    {
        close_fd(_stdin_fd);
        _stdin_closed = true;
    }

    return true;
}

void CgiHandler::terminate()
{
    close_fd(_stdin_fd);
    _stdin_closed = true;

    close_fd(_stdout_fd);
    _stdout_closed = true;

    if (_pid > 0 && !_child_exited)
    {
        kill(_pid, SIGKILL);
        return;
    }

    _pid = -1;
    _child_exited = true;
}


bool CgiHandler::handle_stdin_write(std::string &error_status)
{
    if (_stdin_closed || _stdin_fd == -1)
        return true;

    if (_input_offset >= _input_buffer.size())
    {
        close_fd(_stdin_fd);
        _stdin_closed = true;
        return true;
    }

    std::size_t remaining = _input_buffer.size() - _input_offset;
    std::size_t chunk_size = remaining;

    if (chunk_size > 4096)
        chunk_size = 4096;

    ssize_t bytes_written = write(
        _stdin_fd,
        _input_buffer.data() + _input_offset,
        chunk_size
    );

    if (bytes_written <= 0)
    {
        error_status = "502 Bad Gateway";
        close_fd(_stdin_fd);
        _stdin_closed = true;
        return false;
    }

    _input_offset += static_cast<std::size_t>(bytes_written);

    if (_input_offset >= _input_buffer.size())
    {
        close_fd(_stdin_fd);
        _stdin_closed = true;
    }

    return true;
}

bool CgiHandler::handle_stdout_read(std::string &error_status)
{
    if (_stdout_closed || _stdout_fd == -1)
        return true;

    char buffer[4096];

    ssize_t bytes_read = read(
        _stdout_fd,
        buffer,
        sizeof(buffer)
    );

    if (bytes_read < 0)
    {
        error_status = "502 Bad Gateway";
        close_fd(_stdout_fd);
        _stdout_closed = true;
        return false;
    }

    if (bytes_read == 0)
    {
        close_fd(_stdout_fd);
        _stdout_closed = true;
        return true;
    }

    _output_buffer.append(
        buffer,
        static_cast<std::size_t>(bytes_read)
    );

    return true;
}
bool CgiHandler::check_child(std::string &error_status)
{
    if (_child_exited)
        return true;

    if (_pid <= 0)
    {
        error_status = "500 Internal Server Error";
        return false;
    }

    pid_t result = waitpid(_pid, &_wait_status, WNOHANG);

    if (result == 0)
        return true;

    if (result == _pid)
    {
        _child_exited = true;
        _pid = -1;

        close_fd(_stdin_fd);
        _stdin_closed = true;

        return true;
    }

    error_status = "500 Internal Server Error";
    return false;
}

bool CgiHandler::is_finished() const
{
    return _child_exited &&
           _stdin_closed &&
           _stdout_closed;
}

bool  HttpParser::parse_hex_size(const std::string &size_line, std::size_t &out_chunk_size)
{
    std::size_t chunk_size = 0;

    if (size_line.empty())
        return false;
    for (std::size_t j = 0; j < size_line.size(); ++j)
    {
        unsigned int digit;

        if (size_line[j] >= '0' && size_line[j] <= '9')
            digit = size_line[j] - '0';
        else if (size_line[j] >= 'a' && size_line[j] <= 'f')
            digit = size_line[j] - 'a' + 10;
        else if (size_line[j] >= 'A' && size_line[j] <= 'F')
            digit = size_line[j] - 'A' + 10;
        else
            return false;

        if (chunk_size > (static_cast<std::size_t>(-1) - digit) / 16)
            return false;

        chunk_size = chunk_size * 16 + digit;
    }

    out_chunk_size = chunk_size;
    return true;
}

ChunkedStatus  HttpParser::skip_trailers(const std::string &chunked_body, std::string::size_type &i)
{
    while (true)
    {
        std::string::size_type trailer_end = chunked_body.find("\r\n", i);

        if (trailer_end == std::string::npos)
            return CHUNKED_INCOMPLETE;

        if (trailer_end == i)
            return CHUNKED_COMPLETE;

        std::string trailer = chunked_body.substr(i, trailer_end - i);

        if (trailer.find(':') == std::string::npos)
            return CHUNKED_INVALID;

        i = trailer_end + 2;
    }
}

ChunkedStatus HttpParser::unchunk_body(const std::string &chunked_body, std::string &decoded_body, std::size_t max_body_size)
{
    decoded_body.clear();
    std::string::size_type i = 0;

    while (true)
    {
        std::string::size_type crlf = chunked_body.find("\r\n", i);

        if (crlf == std::string::npos)
            return CHUNKED_INCOMPLETE;

        std::string size_line = chunked_body.substr(i, crlf - i);

        std::string::size_type extension = size_line.find(';');

        if (extension != std::string::npos)
            size_line.erase(extension);

        if (size_line.empty())
            return CHUNKED_INVALID;

        std::size_t chunk_size = 0;
        if (!parse_hex_size(size_line, chunk_size))
            return CHUNKED_INVALID;

        i = crlf + 2;

        if (chunk_size == 0)
        {
            return skip_trailers(chunked_body, i);
        }

        if (max_body_size > 0 &&(decoded_body.size() > max_body_size || chunk_size > max_body_size - decoded_body.size()))
            return CHUNKED_TOO_LARGE;

        if (chunk_size > chunked_body.size() - i)
            return CHUNKED_INCOMPLETE;


        decoded_body.append(chunked_body, i, chunk_size);
        i += chunk_size;

        if (chunked_body.size() - i < 2)
            return CHUNKED_INCOMPLETE;

        if (chunked_body.compare(i, 2, "\r\n") != 0)
            return CHUNKED_INVALID;

        i += 2;
    }
}

void HttpParser::parse_head(const std::string &line, HttpRequest &request, bool &host_found)
{
    std::string::size_type colon = line.find(':');

    if (colon == std::string::npos || colon == 0)
        throw std::runtime_error("Invalid HTTP header");

    std::string name = line.substr(0, colon);
    std::string value = line.substr(colon + 1);

    for (std::size_t i = 0; i < name.size(); ++i)
    {
        name[i] = static_cast<char>(
            std::tolower(static_cast<unsigned char>(name[i]))
        );
    }

    while (!value.empty() && (value[0] == ' ' || value[0] == '\t'))
        value.erase(0, 1);

    while (!value.empty() &&
           (value[value.size() - 1] == ' ' ||
            value[value.size() - 1] == '\t'))
        value.erase(value.size() - 1);

    if (name == "host")
    {
        if (host_found)
            throw std::runtime_error("Duplicate Host header");

        if (value.empty())
            throw std::runtime_error("Empty Host header");
    }

    if (request.headers.find(name) != request.headers.end())
        throw std::runtime_error("Duplicate HTTP header: " + name);

    if (name == "host")
    {
        request.host = value;
        host_found = true;
    }

    request.headers.insert(
        std::map<std::string, std::string>::value_type(name, value)
    );
}

HttpRequest HttpParser::parse(const std::string &raw_request)
{
    HttpRequest request;
    std::istringstream stream(raw_request);
    std::string line;
    std::string extra;
    bool host_found = false;

    std::cout << "line ->" << raw_request << std::endl;
    if (!std::getline(stream, line))
        throw std::runtime_error("Missing request line");
    if (!line.empty() && line[line.size() - 1] == '\r')
        line.erase(line.size() - 1);
    std::istringstream request_line(line);
    if (!(request_line >> request.method >> request.path >> request.version))
        throw std::runtime_error("Invalid HTTP request line");
    
    std::string::size_type query_pos = request.path.find('?');
    if (query_pos != std::string::npos)
    {
        request.query_string = request.path.substr(query_pos + 1);
        request.path = request.path.substr(0, query_pos);
    }
    if (request.path.size() > 2048)
        throw std::runtime_error("414 URI Too Long");
    if (request_line >> extra)
        throw std::runtime_error("Invalid HTTP request line");
    if (request.version != "HTTP/1.0" && request.version != "HTTP/1.1")
        throw std::runtime_error("Unsupported HTTP version");
    if (request.path.empty() || request.path[0] != '/')
        throw std::runtime_error("Invalid request path");
    while (std::getline(stream, line))
    {
        if (!line.empty() && line[line.size() - 1] == '\r')
            line.erase(line.size() - 1);
        if (line.empty())
            break;
        if (line[0] == ' ' || line[0] == '\t')
            throw std::runtime_error("Invalid HTTP header");
        parse_head(line, request, host_found);
    }
    if (request.version == "HTTP/1.1" && !host_found)
        throw std::runtime_error("Missing Host header");
    bool has_transfer_encoding = request.headers.find("transfer-encoding") != request.headers.end();

    bool has_content_length = request.headers.find("content-length") != request.headers.end();

    if (has_transfer_encoding && has_content_length)
        throw std::runtime_error("Transfer-Encoding and Content-Length cannot be used together");
    std::string::size_type body_pos = raw_request.find("\r\n\r\n");
    if (body_pos != std::string::npos)
        request.body = raw_request.substr(body_pos + 4);
    else
    {
        body_pos = raw_request.find("\n\n");
        if (body_pos != std::string::npos)
            request.body = raw_request.substr(body_pos + 2);
    }

    std::map<std::string, std::string>::iterator it =
    request.headers.find("transfer-encoding");

    if (it != request.headers.end())
    {
        std::string transfer_encoding = it->second;

        // Make Transfer-Encoding value lowercase
        for (std::size_t i = 0; i < transfer_encoding.size(); ++i)
        {
            transfer_encoding[i] = static_cast<char>(
                std::tolower(static_cast<unsigned char>(transfer_encoding[i]))
            );
        }

        if (transfer_encoding != "chunked")
            throw std::runtime_error("Unsupported Transfer-Encoding");

        std::string decoded_body;

        ChunkedStatus status = unchunk_body(request.body, decoded_body);

        if (status == CHUNKED_INCOMPLETE)
            throw std::runtime_error("Incomplete chunked body");

        if (status == CHUNKED_INVALID)
            throw std::runtime_error("Invalid chunked body");

        if (status == CHUNKED_TOO_LARGE)
            throw std::runtime_error("Chunked body too large");

        request.body = decoded_body;
    }
    return request;
}

// Canonical spelling for the headers we emit (the map stores lowercase names).
static std::string canonical_header_name(const std::string &lower)
{
    std::string result = lower;
    bool upper_next = true;

    for (std::size_t i = 0; i < result.size(); ++i)
    {
        if (upper_next && result[i] >= 'a' && result[i] <= 'z')
            result[i] = static_cast<char>(result[i] - 'a' + 'A');
        upper_next = (result[i] == '-');
    }
    return result;
}

std::string HttpResponse::build_head(const std::string &version) const
{
    std::ostringstream head;
    const std::string http_version = version.empty() ? "HTTP/1.1" : version;

    head << http_version << " " << _status << " " << _reason << "\r\n";
    head << "Server: " << SERVER_NAME << "\r\n";
    head << "Date: " << http_date(time(NULL)) << "\r\n";

    for (std::map<std::string, std::string>::const_iterator it = _headers.begin();
         it != _headers.end(); ++it)
    {
        // These four are always generated below, never copied.
        if (it->first == "content-length" || it->first == "connection"
            || it->first == "date" || it->first == "server")
            continue;
        head << canonical_header_name(it->first) << ": " << it->second << "\r\n";
    }

    // 1xx / 204 / 304 must not carry a Content-Length or a body.
    if (!status_has_no_body(_status))
        head << "Content-Length: " << number_to_string(_body.size()) << "\r\n";
    head << "Connection: close\r\n";
    head << "\r\n";
    return head.str();
}

std::string HttpResponse::build(const std::string &version) const
{
    if (status_has_no_body(_status))
        return build_head(version);
    return build_head(version) + _body;
}

std::string HttpResponse::reason_phrase(int status)
{
    switch (status)
    {
        case 200: return "OK";
        case 201: return "Created";
        case 202: return "Accepted";
        case 204: return "No Content";
        case 206: return "Partial Content";
        case 301: return "Moved Permanently";
        case 302: return "Found";
        case 303: return "See Other";
        case 304: return "Not Modified";
        case 307: return "Temporary Redirect";
        case 308: return "Permanent Redirect";
        case 400: return "Bad Request";
        case 401: return "Unauthorized";
        case 403: return "Forbidden";
        case 404: return "Not Found";
        case 405: return "Method Not Allowed";
        case 408: return "Request Timeout";
        case 409: return "Conflict";
        case 411: return "Length Required";
        case 413: return "Payload Too Large";
        case 414: return "URI Too Long";
        case 415: return "Unsupported Media Type";
        case 431: return "Request Header Fields Too Large";
        case 500: return "Internal Server Error";
        case 501: return "Not Implemented";
        case 502: return "Bad Gateway";
        case 503: return "Service Unavailable";
        case 504: return "Gateway Timeout";
        case 505: return "HTTP Version Not Supported";
        default: break;
    }
    if (status >= 100 && status < 200)
        return "Informational";
    if (status >= 200 && status < 300)
        return "Success";
    if (status >= 300 && status < 400)
        return "Redirection";
    if (status >= 400 && status < 500)
        return "Client Error";
    return "Server Error";
}

// Looks at the text after the last '.' of the last path component.
std::string HttpResponse::mime_type(const std::string &path)
{
    std::string::size_type slash = path.find_last_of('/');
    std::string name = (slash == std::string::npos) ? path : path.substr(slash + 1);
    std::string::size_type dot = name.find_last_of('.');

    if (dot == std::string::npos || dot + 1 >= name.size())
        return "application/octet-stream";

    const std::string extension = to_lower(name.substr(dot + 1));

    for (std::size_t i = 0; MIME_TABLE[i].extension != NULL; ++i)
    {
        if (extension == MIME_TABLE[i].extension)
            return MIME_TABLE[i].type;
    }
    return "application/octet-stream";
}

// RFC 7231 IMF-fixdate, always in English regardless of the locale.
std::string HttpResponse::http_date(time_t when)
{
    static const char *days[] = {"Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"};
    static const char *months[] = {"Jan", "Feb", "Mar", "Apr", "May", "Jun",
                                   "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"};
    struct tm parts;

    if (gmtime_r(&when, &parts) == NULL)
        return "Thu, 01 Jan 1970 00:00:00 GMT";

    std::ostringstream date;
    date << days[parts.tm_wday] << ", " << two_digits(parts.tm_mday) << " "
         << months[parts.tm_mon] << " " << (parts.tm_year + 1900) << " "
         << two_digits(parts.tm_hour) << ":" << two_digits(parts.tm_min) << ":"
         << two_digits(parts.tm_sec) << " GMT";
    return date.str();
}

std::string HttpResponse::html_escape(const std::string &text)
{
    std::string result;

    for (std::size_t i = 0; i < text.size(); ++i)
    {
        switch (text[i])
        {
            case '&': result += "&amp;"; break;
            case '<': result += "&lt;"; break;
            case '>': result += "&gt;"; break;
            case '"': result += "&quot;"; break;
            case '\'': result += "&#39;"; break;
            default: result += text[i]; break;
        }
    }
    return result;
}

// "404 Not Found" + body  ->  HttpResponse.
// Content-Type is guessed: markup stays HTML, anything else is plain text.
HttpResponse HttpResponse::from_status_line(const std::string &status_line, const std::string &body)
{
    int code = std::atoi(status_line.c_str());

    if (code < 100 || code > 599)
        code = 500;

    HttpResponse response(code);
    std::string::size_type space = status_line.find(' ');

    if (space != std::string::npos && space + 1 < status_line.size())
        response.set_status(code, status_line.substr(space + 1));

    if (!body.empty())
    {
        if (body[0] == '<')
            response.set_body(body, "text/html; charset=UTF-8");
        else
            response.set_body(body, "text/plain; charset=UTF-8");
    }
    return response;
}

#include "Router.hpp"

namespace
{
    int hex_value(char c)
    {
        if (c >= '0' && c <= '9')
            return c - '0';
        if (c >= 'a' && c <= 'f')
            return c - 'a' + 10;
        if (c >= 'A' && c <= 'F')
            return c - 'A' + 10;
        return -1;
    }

    bool is_unreserved(char c)
    {
        return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
            || (c >= '0' && c <= '9') || c == '-' || c == '.'
            || c == '_' || c == '~' || c == '/';
    }

    // Decodes %XX sequences exactly once. Fails on bad hex or an encoded NUL.
    bool percent_decode(const std::string &raw, std::string &decoded)
    {
        decoded.clear();
        for (std::size_t i = 0; i < raw.size(); ++i)
        {
            if (raw[i] != '%')
            {
                if (raw[i] == '\0')
                    return false;
                decoded += raw[i];
                continue;
            }
            if (i + 2 >= raw.size())
                return false;
            int high = hex_value(raw[i + 1]);
            int low = hex_value(raw[i + 2]);
            if (high < 0 || low < 0)
                return false;
            char value = static_cast<char>(high * 16 + low);
            if (value == '\0')
                return false;
            decoded += value;
            i += 2;
        }
        return true;
    }

    // Joins root and relative part with exactly one '/' between them.
    std::string join_path(const std::string &root, const std::string &relative)
    {
        std::string result = root;
        std::size_t start = 0;

        while (start < relative.size() && relative[start] == '/')
            ++start;
        while (result.size() > 1 && result[result.size() - 1] == '/')
            result.erase(result.size() - 1);
        if (result.empty() || result[result.size() - 1] != '/')
            result += '/';
        result += relative.substr(start);
        return result;
    }

NormalizeStatus Router::normalize_uri(const std::string &raw, std::string &normalized)
{
    std::string decoded;

    normalized = "/";
    if (raw.empty() || raw[0] != '/')
        return NORMALIZE_BAD_REQUEST;
    if (!percent_decode(raw, decoded))
        return NORMALIZE_BAD_REQUEST;

    std::vector<std::string> segments;
    std::size_t start = 0;

    while (start <= decoded.size())
    {
        std::size_t end = decoded.find('/', start);
        if (end == std::string::npos)
            end = decoded.size();

        const std::string segment = decoded.substr(start, end - start);

        if (segment == "..")
        {
            if (segments.empty())
                return NORMALIZE_FORBIDDEN;
            segments.pop_back();
        }
        else if (!segment.empty() && segment != ".")
            segments.push_back(segment);
        start = end + 1;
    }

    normalized.clear();
    for (std::size_t i = 0; i < segments.size(); ++i)
    {
        normalized += '/';
        normalized += segments[i];
    }
    if (normalized.empty())
        normalized = "/";

    // A trailing slash ("/dir/", "/dir/.", "/dir/..") still means "directory".
    const std::string::size_type last_slash = decoded.find_last_of('/');
    const std::string last = decoded.substr(last_slash + 1);
    if (normalized != "/" && (last.empty() || last == "." || last == ".."))
        normalized += '/';
    return NORMALIZE_OK;
}

std::string Router::encode_uri(const std::string &path)
{
    static const char *hex = "0123456789ABCDEF";
    std::string result;

    for (std::size_t i = 0; i < path.size(); ++i)
    {
        if (is_unreserved(path[i]))
            result += path[i];
        else
        {
            unsigned char byte = static_cast<unsigned char>(path[i]);
            result += '%';
            result += hex[byte >> 4];
            result += hex[byte & 0x0F];
        }
    }
    return result;
}

const LocationConfig *Router::find_location(const std::string &uri) const
{
    const LocationConfig *best = NULL;

    for (std::size_t i = 0; i < _conf.locations.size(); ++i)
    {
        const std::string &prefix = _conf.locations[i].path;
        bool matches = false;

        if (prefix.empty())
            continue;

        if (prefix == "/")
            matches = true;
        else if (prefix[prefix.size() - 1] == '/')
        {
            // "/images/" matches "/images/x" and also "/images" itself.
            matches = uri.compare(0, prefix.size(), prefix) == 0
                || uri == prefix.substr(0, prefix.size() - 1);
        }
        else if (uri.size() >= prefix.size() && uri.compare(0, prefix.size(), prefix) == 0)
            matches = (uri.size() == prefix.size() || uri[prefix.size()] == '/');

        if (matches && (best == NULL || prefix.size() > best->path.size()))
            best = &_conf.locations[i];
    }
    return best;
}

        return location->root;
    return _conf.root;
}

// The part of the URI that matched the location is replaced by the root, so
// "location /kapouet { root /tmp/www; }" maps /kapouet/a/b to /tmp/www/a/b.
std::string Router::build_path(const std::string &uri, const LocationConfig *location) const
{
    std::string relative = uri;

    if (location != NULL && location->path != "/")
    {
        std::string prefix = location->path;

        if (prefix[prefix.size() - 1] == '/')
            prefix.erase(prefix.size() - 1);
        relative = uri.substr(prefix.size());
    }
    return join_path(get_root(location), relative);
}

bool Router::is_method_allowed(const LocationConfig *location, const std::string &method) const
{
    if (method != "GET" && method != "POST" && method != "DELETE")
        return false;
    if (location == NULL || location->allowed_methods.empty())
        return true;
    for (std::size_t i = 0; i < location->allowed_methods.size(); ++i)
    {
        if (location->allowed_methods[i] == method)
            return true;
    }
    return false;
}

std::string Router::allowed_methods_header(const LocationConfig *location) const
{
    static const char *all[] = {"GET", "POST", "DELETE"};
    std::string result;

    for (std::size_t i = 0; i < 3; ++i)
    {
        if (!is_method_allowed(location, all[i]))
            continue;
        if (!result.empty())
            result += ", ";
        result += all[i];
    }
    return result;
}

RouteResult Router::route(const HttpRequest &request) const
{
    RouteResult result;

    switch (normalize_uri(request.path, result.uri))
    {
        case NORMALIZE_BAD_REQUEST:
            result.status = ROUTE_BAD_REQUEST;
            return result;
        case NORMALIZE_FORBIDDEN:
            result.status = ROUTE_FORBIDDEN;
            return result;
        case NORMALIZE_OK:
            break;
    }

    result.location = find_location(result.uri);
    result.root = get_root(result.location);
    result.fs_path = build_path(result.uri, result.location);

    // "return" wins over everything else, exactly like nginx.
    if (result.location != NULL && result.location->return_code != 0)
    {
        result.status = ROUTE_REDIRECT;
        result.redirect_code = result.location->return_code;
        result.redirect_url = result.location->return_url;
        return result;
    }

    if (!is_method_allowed(result.location, request.method))
    {
        result.status = ROUTE_METHOD_NOT_ALLOWED;
        result.allow = allowed_methods_header(result.location);
        return result;
    }
    return result;
}

void Server::set_nonblocking(int fd)
{
    int flags = fcntl(fd, F_GETFL, 0);

    if (flags == -1)
        throw std::runtime_error("Failed to get socket flags");

    if (fcntl(fd, F_SETFL, flags | O_NONBLOCK) == -1)
        throw std::runtime_error("Failed to set non-blocking mode");
}

void Server::create_socket()
{
    for (std::size_t i = 0; i < _listens.size(); ++i)
    {
        int server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd == -1)
            throw std::runtime_error("Failed to create socket");
        int reuse = 1;
        if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse)) == -1)
        {
            close(server_fd);
            throw std::runtime_error("Failed to set SO_REUSEADDR");
        }
        try
        {
            set_nonblocking(server_fd);
        }
        catch (...)
        {
            close(server_fd);
            server_fd = -1;
            throw;
        }
        _server_fds.push_back(server_fd);
        std::cout << "Created socket FD: " << server_fd << " for port: " << _listens[i].port << std::endl;
    }

}


void Server::listen_socket()
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        if (listen(_server_fds[i], 128) == -1)
            throw std::runtime_error("Failed to listen on socket");
    }
}

void Server::bind_socket()
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        struct sockaddr_in address;

        std::memset(&address, 0, sizeof(address));
        address.sin_family = AF_INET;
        address.sin_port = htons(_listens[i].port);
        if (inet_pton(AF_INET,_listens[i].ip.c_str(), &address.sin_addr) != 1)
            throw std::runtime_error("Invalid IPv4 address");
        if (bind(_server_fds[i], reinterpret_cast<struct sockaddr *>(&address), sizeof(address)) == -1)
            throw std::runtime_error("Failed to bind socket");
        std::cout << "Bind successful: " << _listens[i].ip << ":" << _listens[i].port << std::endl;
    }
}

int Server::accept_connection(int server_fd)
{
    int client_fd = accept(server_fd, NULL, NULL);

    if (client_fd == -1)
    {
        if (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK || errno == ECONNABORTED)
            return -1;
        throw std::runtime_error("Failed to accept connection");
    }
    try
    {
        set_nonblocking(client_fd);
    }
    catch (...)
    {
        close(client_fd);
        return -1;
    }
    return client_fd;
}

void Server::handle_delete(const HttpRequest &request, const ServerConfig &conf,std::string &status, std::string &body)
{
    const LocationConfig *location = find_location(request.path, conf);

    if (location != NULL && !location->allowed_methods.empty())
    {
        bool allowed = false;

        for (std::size_t i = 0; i < location->allowed_methods.size(); ++i)
        {
            if (location->allowed_methods[i] == "DELETE")
                allowed = true;
        }
        if (!allowed)
        {
            status = "405 Method Not Allowed";
            body = "405 Method Not Allowed\n";
            return;
        }
    }

    if (request.path.empty() || request.path[0] != '/')
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    std::string root = conf.root;
    if (location != NULL && !location->root.empty())
        root = location->root;

    if (root.empty())
    {
        status = "500 Internal Server Error";
        body = "Invalid server root\n";
        return;
    }

    const std::string file_path = build_file_path(request, location, conf);

    std::string normalized_root = root;
    if (normalized_root[normalized_root.size() - 1] != '/')
        normalized_root += '/';

    if (file_path.compare(0, normalized_root.size(), normalized_root) != 0)
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    // C filesystem calls must see the complete mapped path.
    if (file_path.find('\0') != std::string::npos)
    {
        status = "400 Bad Request";
        body = "400 Bad Request\n";
        return;
    }

    // Reject parent components in the mapped suffix without decoding the URI.
    const std::string relative_path = file_path.substr(normalized_root.size());
    std::size_t start = 0;
    while (start < relative_path.size())
    {
        std::size_t end = relative_path.find('/', start);
        if (end == std::string::npos)
            end = relative_path.size();
        if (relative_path.substr(start, end - start) == "..")
        {
            status = "403 Forbidden";
            body = "403 Forbidden\n";
            return;
        }
        start = end + 1;
    }

    if (file_path.find_first_not_of('/') == std::string::npos)
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    // Inspect every prefix, including the configured root, without following links.
    // This check does not prevent another process from replacing a directory.
    struct stat info;
    bool checked = false;
    start = 0;
    while (start < file_path.size())
    {
        std::size_t end = file_path.find('/', start);
        if (end == std::string::npos)
            end = file_path.size();
        if (end == start)
        {
            start = end + 1;
            continue;
        }
        const std::string prefix = file_path.substr(0, end);
        if (lstat(prefix.c_str(), &info) == -1)
        {
            checked = false;
            break;
        }
        if (S_ISLNK(info.st_mode))
        {
            status = "403 Forbidden";
            body = "403 Forbidden\n";
            return;
        }
        if (end < file_path.size() && !S_ISDIR(info.st_mode))
        {
            status = "404 Not Found";
            body = "404 Not Found\n";
            return;
        }
        checked = true;
        start = end + 1;
    }
    if (!checked)
    {
        if (errno == ENOENT || errno == ENOTDIR)
            status = "404 Not Found";
        else if (errno == EACCES || errno == EPERM)
            status = "403 Forbidden";
        else
            status = "500 Internal Server Error";
        body = status + "\n";
        return;
    }

    if (!S_ISREG(info.st_mode))
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    if (unlink(file_path.c_str()) == -1)
    {
        if (errno == ENOENT || errno == ENOTDIR)
            status = "404 Not Found";
        else if (errno == EACCES || errno == EPERM || errno == EROFS)
            status = "403 Forbidden";
        else
            status = "500 Internal Server Error";

        body = status + "\n";
        return;
    }

    status = "204 No Content";
    body.clear();
}

std::string Server::build_response(const HttpRequest &request,const ServerConfig &conf)
{
    const Router router(conf);
    const StaticFileHandler files(conf, router);
    const RouteResult route = router.route(request);
    HttpResponse response;

    switch (route.status)
    {
        case ROUTE_BAD_REQUEST:
            response = files.make_error(400);
            break;
        case ROUTE_FORBIDDEN:
            response = files.make_error(403);
            break;
        case ROUTE_REDIRECT:
            response = files.make_redirect(route.redirect_code, route.redirect_url);
            break;
        case ROUTE_METHOD_NOT_ALLOWED:
            response = files.make_error(405);
            response.set_header("Allow", route.allow);
            break;
        case ROUTE_OK:
        {
            // Handlers receive the decoded, normalized path.
            HttpRequest routed = request;
            std::string status;
            std::string body;

            routed.path = route.uri;
            if (request.method == "GET")
                response = files.handle_get(routed, route);
            else if (request.method == "POST")
            {
                handle_post(routed, status, body);
                response = finish_handler_response(files, status, body);
            }
            else
            {
                handle_delete(routed, conf, status, body);
                response = finish_handler_response(files, status, body);
            }
            break;
        }
    }
    return response.build(request.version);
}

bool Server::get_content_length(const std::string &headers_part,std::size_t &content_length) const
{
    content_length = 0;
    bool found = false;
    std::string::size_type line_start = 0;

    while (line_start < headers_part.size())
    {
        std::string::size_type line_end = headers_part.find("\r\n", line_start);

        if (line_end == std::string::npos)
            line_end = headers_part.size();

        std::string::size_type colon = headers_part.find(':', line_start);

        if (colon != std::string::npos && colon < line_end)
        {
            std::string name = headers_part.substr(line_start, colon - line_start);

            for (std::string::size_type i = 0; i < name.size(); ++i)
            {
                if (name[i] >= 'A' && name[i] <= 'Z')
                    name[i] = name[i] - 'A' + 'a';
            }

            if (name == "content-length")
            {
                if (found)
                    return false;

                found = true;

                std::string::size_type start = colon + 1;

                while (start < line_end &&
                       (headers_part[start] == ' ' ||
                        headers_part[start] == '\t'))
                    ++start;

                std::string::size_type end = line_end;

                while (end > start && (headers_part[end - 1] == ' ' || headers_part[end - 1] == '\t'))
                    --end;

                if (start == end)
                    return false;

                std::size_t value = 0;

                for (std::string::size_type i = start; i < end; ++i)
                {
                    if (headers_part[i] < '0' ||
                        headers_part[i] > '9')
                        return false;

                    std::size_t digit =
                        static_cast<std::size_t>(headers_part[i] - '0');

                    if (value >
                        (std::numeric_limits<std::size_t>::max() - digit) / 10)
                        return false;

                    value = value * 10 + digit;
                }

                content_length = value;
            }
        }

        if (line_end == headers_part.size())
            break;

        line_start = line_end + 2;
    }

    return true;
}



bool Server::check_max_body_size(ClientConnection &client, std::size_t content_length, const ServerConfig &conf)
{
    if (conf.client_max_body_size > 0 && content_length > conf.client_max_body_size)
    {
        std::cerr << "Error: Request body exceeds client_max_body_size" << std::endl;
        HttpRequest dummy_req;
        dummy_req.version = "HTTP/1.1";
        std::string status = "413 Payload Too Large";
        std::string body = "413 Payload Too Large\n";
        client.response_buffer = create_response_data(dummy_req, status, body) + body;
        client.request_complete = true;
        client.response_ready = true;
        return true;
    }
    return false;
}

    {

        std::cout << "Method:  " << request.method << std::endl;
        std::cout << "Path:    " << request.path << std::endl;

        std::cout << "Query: [" << request.query_string << "]" << std::endl;
        std::cout << "========================" << std::endl;

        client.response_buffer = build_response(request, conf);
        client.request_complete = true;
        client.response_ready = true;
    }
    catch(const std::exception &e)
    {
        std::cerr << "Parse error: " << e.what() << std::endl;
        HttpRequest request;
        request.version = "HTTP/1.1";

        std::string status = "400 Bad Request";

        {
            status = err_msg;
        }

        client.response_buffer = create_response_data(request, status, body) + body;
        client.request_complete = true;
        client.response_ready = true;
    }
}


bool Server::read_client_data(ClientConnection &client, bool &stop_processing)
{
    const std::size_t MAX_HEADER_SIZE = 32 * 1024;

    char buffer[4096];
    stop_processing = false;

    ssize_t bytes_read = recv(client.fd, buffer, sizeof(buffer), 0);

    if (bytes_read == 0)
        return false;

    if (bytes_read < 0)
    {
        stop_processing = true;
        return true;
    }

    client.last_activity = time(NULL);

    client.request_buffer.append(
        buffer,
        static_cast<std::string::size_type>(bytes_read)
    );

    std::string::size_type header_end = client.request_buffer.find("\r\n\r\n");

    if (header_end == std::string::npos)
    {
        if (client.request_buffer.size() > MAX_HEADER_SIZE)
        {
            client.header_too_large = true;
            stop_processing = true;
        }
    }
    else if (header_end + 4 > MAX_HEADER_SIZE)
    {
        client.header_too_large = true;
        stop_processing = true;
    }

    return true;
}

bool Server::is_chunked(const std::string &headers_part) const
{
    std::string::size_type line_start = 0;

    while (line_start < headers_part.size())
    {
        std::string::size_type line_end = headers_part.find("\r\n", line_start);

        if (line_end == std::string::npos)
            line_end = headers_part.size();

        std::string::size_type colon = headers_part.find(':', line_start);

        if (colon != std::string::npos && colon < line_end)
        {
            std::string name = headers_part.substr(line_start, colon - line_start);

            for (std::string::size_type i = 0; i < name.size(); ++i)
            {
                if (name[i] >= 'A' && name[i] <= 'Z')
                    name[i] = name[i] - 'A' + 'a';
            }

            if (name == "transfer-encoding")
            {
                std::string::size_type token_start = colon + 1;

                while (token_start < line_end)
                {
                    std::string::size_type comma_pos =
                        headers_part.find(',', token_start);

                    if (comma_pos == std::string::npos || comma_pos > line_end)
                        comma_pos = line_end;

                    std::string::size_type token_end = comma_pos;

                    while (token_start < token_end && (headers_part[token_start] == ' ' || headers_part[token_start] == '\t'))
                        ++token_start;

                    while (token_end > token_start &&
                           (headers_part[token_end - 1] == ' ' ||
                            headers_part[token_end - 1] == '\t'))
                        --token_end;

                    std::string token =
                        headers_part.substr(
                            token_start, token_end - token_start);

                    for (std::string::size_type i = 0;
                         i < token.size(); ++i)
                    {
                        if (token[i] >= 'A' && token[i] <= 'Z')
                            token[i] = token[i] - 'A' + 'a';
                    }

                    if (token == "chunked")
                        return true;

                    if (comma_pos == line_end)
                        break;

                    token_start = comma_pos + 1;
                }
            }
        }

        if (line_end == headers_part.size())
            break;

        line_start = line_end + 2;
    }

    return false;
}


void Server::handle_large_header(ClientConnection &client)
{
    HttpRequest dummy_req;
    dummy_req.version = "HTTP/1.1";

    std::string status = "431 Request Header Fields Too Large";
    std::string body = status + "\n";

    client.response_buffer =
        create_response_data(dummy_req, status, body) + body;

    client.request_complete = true;
    client.response_ready = true;
}

bool Server::handle_client_event(ClientConnection &client, HttpParser &http_parser)
{
    bool stop_processing = false;
    if (!read_client_data(client, stop_processing))
        return false;
    if (client.header_too_large)
    {
        handle_large_header(client);
        return true;
    }
    if (stop_processing)
        return true;

    std::string::size_type header_end = client.request_buffer.find("\r\n\r\n");
    if (header_end == std::string::npos)
        return true; 
    const ServerConfig &conf = get_server_config(client);

    std::string headers_part = client.request_buffer.substr(0, header_end);
    std::string::size_type body_start = header_end + 4;

    if (is_chunked(headers_part))
    {
        std::string decoded_body;

        ChunkedStatus chunked_status = HttpParser::unchunk_body(client.request_buffer.substr(body_start), decoded_body, conf.client_max_body_size);

        if (chunked_status == CHUNKED_INCOMPLETE)
            return true;

        if (chunked_status == CHUNKED_INVALID)
        {
            parse_and_respond(client, http_parser, conf);
            return true;
        }
        if (chunked_status == CHUNKED_TOO_LARGE)
        {
            HttpRequest dummy_req;
            dummy_req.version = "HTTP/1.1";

            std::string status = "413 Payload Too Large";
            std::string body = status + "\n";

            client.response_buffer =
                create_response_data(dummy_req, status, body) + body;

            client.request_complete = true;
            client.response_ready = true;
            return true;
        }

        if (check_max_body_size(client, decoded_body.size(), conf))
            return true;
    }
    else
    {
        std::size_t content_length = 0;

        if (!get_content_length(headers_part, content_length))
        {
            HttpRequest dummy_req;
            dummy_req.version = "HTTP/1.1";

            std::string status = "400 Bad Request";
            std::string body = status + "\n";

            client.response_buffer =
                create_response_data(dummy_req, status, body) + body;

            client.request_complete = true;
            client.response_ready = true;
            return true;
        }

        if (check_max_body_size(client, content_length, conf))
            return true;

        if (client.request_buffer.size() - body_start < content_length)
            return true;
    }

    parse_and_respond(client, http_parser, conf);
    return true;
}

const ServerConfig &Server::get_server_config(const ClientConnection &client) const
{
    if (client.listener_index >= _listen_config_indices.size())
        throw std::runtime_error("Invalid listener index");

    std::size_t config_index = _listen_config_indices[client.listener_index];
    if (config_index >= _configs.size())
        throw std::runtime_error("Invalid config index");

    return _configs[config_index];
}



bool Server::handle_client_write(ClientConnection &client)
{
    if (client.sent_bytes >= client.response_buffer.size())
        return false;

    ssize_t bytes_sent = send(client.fd, client.response_buffer.c_str() + client.sent_bytes,
        client.response_buffer.size() - client.sent_bytes,
        0);

    if (bytes_sent < 0)
        return false;

    if (bytes_sent == 0)
        return false;

    client.sent_bytes += static_cast<std::size_t>(bytes_sent);
    client.last_activity = time(NULL);

    if (client.sent_bytes < client.response_buffer.size())
        return true;
    std::cout << "Response fully sent to client FD: " << client.fd << std::endl;

    bool close_connection =(client.response_buffer.find("Connection: close") != std::string::npos);
    if (close_connection)
        return false;
    client.reset();
    return true;
}  

const ServerConfig *Server::find_server_config(const std::string &host) const
{
    if (_configs.empty())
        return NULL;
    
    std::string host_name = host;

    std::string::size_type colon = host_name.find(':');
    if (colon != std::string::npos)
        host_name.erase(colon);
    for (std::size_t i = 0; i < host_name.size(); ++i)
    {
        host_name[i] = static_cast<char>(std::tolower(static_cast<unsigned char>(host_name[i])));
    }
    for (std::size_t i = 0; i < _configs.size(); ++i)
    {
        if (_configs[i].server_name.empty())
            continue;

        std::string configured_name = _configs[i].server_name;

        for (std::size_t j = 0; j < configured_name.size(); ++j)
        {
            configured_name[j] = static_cast<char>(std::tolower(static_cast<unsigned char>(configured_name[j])));
        }
        if (configured_name == host_name)
            return &_configs[i];
    }
    return  NULL;
}

std::vector<struct pollfd> Server::build_poll_fds(const std::vector<ClientConnection> &clients) const
{
    std::vector<struct pollfd> fds;

    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        struct pollfd server_poll;

        server_poll.fd = _server_fds[i];
        server_poll.events = POLLIN;
        server_poll.revents = 0;

        fds.push_back(server_poll);
    }

    for (std::size_t i = 0; i < clients.size(); ++i)
    {
        struct pollfd client_poll;

        client_poll.fd = clients[i].fd;

        if (clients[i].cgi_pending)
            client_poll.events = 0;
        else if (clients[i].response_ready)
            client_poll.events = POLLOUT;
        else
            client_poll.events = POLLIN;

        client_poll.revents = 0;

        fds.push_back(client_poll);
    }
    for (std::size_t i = 0; i < _cgi_processes.size(); ++i)
    {
        CgiHandler *cgi = _cgi_processes[i];

        int stdin_fd = -1;
        int stdout_fd = -1;

        if (cgi != NULL)
        {
            stdin_fd = cgi->get_stdin_fd();
            stdout_fd = cgi->get_stdout_fd();
        }

        struct pollfd input_poll;

        input_poll.fd = stdin_fd;
        input_poll.events = 0;
        input_poll.revents = 0;

        if (stdin_fd != -1)
            input_poll.events = POLLOUT;

        fds.push_back(input_poll);

        struct pollfd output_poll;

        output_poll.fd = stdout_fd;
        output_poll.events = 0;
        output_poll.revents = 0;

        if (stdout_fd != -1)
            output_poll.events = POLLIN;

        fds.push_back(output_poll);
    }
    return fds;
}




void Server::accept_ready_connections(const std::vector<struct pollfd> &fds, std::vector<ClientConnection> &clients)
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        if (_server_fds[i] == -1)
            continue;

        if (fds[i].revents & POLLIN)
        {
            int client_fd = accept_connection(_server_fds[i]);

            if (client_fd != -1)
            {
                std::cout << "Accepted client: " << client_fd << std::endl;
                clients.push_back(ClientConnection(client_fd, i));
            }
        }

        if (fds[i].revents & (POLLERR | POLLHUP | POLLNVAL))
        {
            std::cerr << "Error: Listening socket FD " << _server_fds[i]
                      << " encountered error/hangup/invalid state." << std::endl;
            close(_server_fds[i]);
            _server_fds[i] = -1;
        }
    }
}


int Server::wait_for_events(std::vector<struct pollfd> &fds, int timeout_ms) const
{
    if (fds.empty())
        return 0;

    int result = poll(&fds[0], fds.size(),timeout_ms);

    if (result == -1)
    {
        if (errno == EINTR)
            return -1;

        throw std::runtime_error("poll failed");
    }
    return result;
}

void Server::process_client_events(const std::vector<struct pollfd> &fds,std::vector<ClientConnection> &clients,HttpParser &parser,std::size_t polled_client_count)
{
    for (std::size_t i = _server_fds.size(); i < _server_fds.size() + polled_client_count && i < fds.size(); ++i)
    {
        int client_fd = fds[i].fd;
        short events = fds[i].revents;

        if (events == 0)
            continue;

        std::vector<ClientConnection>::iterator it = clients.begin();

        while (it != clients.end() && it->fd != client_fd)
            ++it;

        if (it == clients.end())
            continue;

        if (events & POLLNVAL)
        {
            remove_client(client_fd, clients);
            continue;
        }

        bool keep_client = true;

        if (events & POLLIN)
            keep_client = handle_client_event(*it, parser);

        if (keep_client && (events & POLLOUT))
            keep_client = handle_client_write(*it);

        if (events & POLLERR)
            keep_client = false;

        if ((events & POLLHUP) && !(events & POLLIN) && !(events & POLLOUT))
            keep_client = false;

        if (!keep_client)
            remove_client(client_fd, clients);
    }
}


void Server::check_timeouts(std::vector<ClientConnection> &clients, time_t timeout_seconds)
{
    time_t current_time = time(NULL);
    for(std::size_t i = 0; i < clients.size(); )
    {
        if(current_time - clients[i].last_activity > timeout_seconds)
        {
            std::cout << "Client FD " << clients[i].fd << " timed out. Closing connection." << std::endl;
            close(clients[i].fd);
            clients.erase(clients.begin() + i);
        }
        else
            ++i;
    }
}

void Server::run(HttpParser &parser)
{
    std::vector<ClientConnection> clients;
    const time_t TIMEOUT_SECONDS = 5;

    while (true)
    {
        check_timeouts(clients, TIMEOUT_SECONDS);

        std::size_t polled_client_count = clients.size();

        std::vector<struct pollfd> fds;
        fds = build_poll_fds(clients);

        int result = wait_for_events(fds, 1000);

        if (result <= 0)
            continue;

        accept_ready_connections(fds, clients);

        process_client_events(
            fds,
            clients,
            parser,
            polled_client_count
        );
    }
}


Server::~Server()
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        if (_server_fds[i] != -1)
            close(_server_fds[i]);
    }
}

namespace
{
    struct DirEntry
    {
        std::string name;
        bool is_dir;
        off_t size;
        time_t mtime;
    };

    // Directories first, then case-sensitive alphabetical order.
    bool entry_less(const DirEntry &a, const DirEntry &b)
    {
        if (a.is_dir != b.is_dir)
            return a.is_dir;
        return a.name < b.name;
    }

    std::string size_to_string(off_t size)
    {
        std::ostringstream ss;

        ss << static_cast<unsigned long long>(size);
        return ss.str();
    }
}

int StaticFileHandler::status_from_errno(int error_number)
{
    switch (error_number)
    {
        case ENOENT:
        case ENOTDIR:
        case ENAMETOOLONG:
            return 404;
        case EACCES:
        case EPERM:
            return 403;
        default:
            return 500;
    }
}

bool StaticFileHandler::read_file(const std::string &path, std::string &content)
{
    std::ifstream file(path.c_str(), std::ios::in | std::ios::binary);

    if (!file)
        return false;

    std::ostringstream buffer;

    buffer << file.rdbuf();
    if (file.bad())
        return false;
    content = buffer.str();
    return true;
}

HttpResponse StaticFileHandler::make_redirect(int status, const std::string &location) const
{
    return HttpResponse::redirect(status, location);
}

/*
** The configured error_page is a URI ("/errors/404.html"). It is routed like
** a normal request (location + root) and read from disk. If anything goes
** wrong - not configured, file missing, unreadable - the built-in page is used,
** so building an error can never fail or loop.
*/
HttpResponse StaticFileHandler::make_error(int status) const
{
    std::map<int, std::string>::const_iterator it = _conf.error_pages.find(status);

    if (it != _conf.error_pages.end())
    {
        std::string uri;

        if (Router::normalize_uri(it->second, uri) == NORMALIZE_OK)
        {
            const LocationConfig *location = _router.find_location(uri);
            const std::string path = _router.build_path(uri, location);
            struct stat info;
            std::string content;

            if (stat(path.c_str(), &info) == 0 && S_ISREG(info.st_mode)
                && read_file(path, content))
            {
                HttpResponse response(status);

                response.set_body(content, HttpResponse::mime_type(path));
                return response;
            }
        }
    }
    return HttpResponse::default_error(status);
}

HttpResponse StaticFileHandler::serve_file(const std::string &path) const
{
    struct stat info;

    if (stat(path.c_str(), &info) != 0)
        return make_error(status_from_errno(errno));
    if (!S_ISREG(info.st_mode))
        return make_error(403);
    if (access(path.c_str(), R_OK) != 0)
        return make_error(403);

    std::string content;

    if (!read_file(path, content))
        return make_error(500);

    HttpResponse response(200);

    response.set_body(content, HttpResponse::mime_type(path));
    response.set_header("Last-Modified", HttpResponse::http_date(info.st_mtime));
    return response;
}

HttpResponse StaticFileHandler::build_autoindex(const std::string &dir_path, const std::string &uri) const
{
    DIR *dir = opendir(dir_path.c_str());

    if (dir == NULL)
        return make_error(status_from_errno(errno));

    std::vector<DirEntry> entries;
    std::string base = dir_path;

    if (base.empty() || base[base.size() - 1] != '/')
        base += '/';

    struct dirent *item;
    while ((item = readdir(dir)) != NULL)
    {
        const std::string name = item->d_name;
        struct stat info;
        DirEntry entry;

        if (name == "." || name == "..")
            continue;
        if (stat((base + name).c_str(), &info) != 0)
            continue;               // broken symlink or file removed meanwhile
        entry.name = name;
        entry.is_dir = S_ISDIR(info.st_mode);
        entry.size = info.st_size;
        entry.mtime = info.st_mtime;
        entries.push_back(entry);
    }
    closedir(dir);
    std::sort(entries.begin(), entries.end(), entry_less);

    std::ostringstream html;
    const std::string title = "Index of " + HttpResponse::html_escape(uri);

    html << "<!DOCTYPE html>\n<html>\n<head>\n<meta charset=\"UTF-8\">\n<title>"
         << title << "</title>\n</head>\n<body>\n<h1>" << title << "</h1>\n<hr>\n<pre>\n";
    if (uri != "/")
        html << "<a href=\"../\">../</a>\n";
    for (std::size_t i = 0; i < entries.size(); ++i)
    {
        const std::string shown = entries[i].name + (entries[i].is_dir ? "/" : "");
        std::string line = "<a href=\"" + Router::encode_uri(shown) + "\">"
            + HttpResponse::html_escape(shown) + "</a>";
        std::string date = HttpResponse::http_date(entries[i].mtime);

        // Simple column alignment: pad the visible name to 40 characters.
        if (shown.size() < 40)
            line += std::string(40 - shown.size(), ' ');
        else
            line += ' ';
        html << line << date << "  "
             << (entries[i].is_dir ? std::string("-") : size_to_string(entries[i].size))
             << "\n";
    }
    html << "</pre>\n<hr>\n</body>\n</html>\n";

    HttpResponse response(200);

    response.set_body(html.str(), "text/html; charset=UTF-8");
    return response;
}

/*
** Directory handling order:
**   1. no trailing slash        -> 301 to the same URI with a slash
**   2. index file exists        -> serve it
**   3. autoindex on             -> generated listing
**   4. otherwise                -> 403
*/
HttpResponse StaticFileHandler::serve_directory(const HttpRequest &request, const RouteResult &route) const
{
    if (route.uri[route.uri.size() - 1] != '/')
    {
        std::string location = Router::encode_uri(route.uri) + "/";

        if (!request.query_string.empty())
            location += "?" + request.query_string;
        return make_redirect(301, location);
    }

    const std::string index = _router.get_index(route.location);

    if (!index.empty())
    {
        std::string index_path = route.fs_path;
        struct stat info;

        if (index_path[index_path.size() - 1] != '/')
            index_path += '/';
        index_path += index;
        const int result = stat(index_path.c_str(), &info);
        const int error_number = errno;

        if (result == 0 && S_ISREG(info.st_mode))
            return serve_file(index_path);
        if (result != 0 && error_number == EACCES)
            return make_error(403);
    }

    if (route.location != NULL && route.location->autoindex)
        return build_autoindex(route.fs_path, route.uri);
    return make_error(403);
}

HttpResponse StaticFileHandler::handle_get(const HttpRequest &request, const RouteResult &route) const
{
    struct stat info;

    if (stat(route.fs_path.c_str(), &info) != 0)
        return make_error(status_from_errno(errno));
    if (S_ISDIR(info.st_mode))
        return serve_directory(request, route);
    return serve_file(route.fs_path);
}

void Tokenizer::add_token_word(std::string &word)
{
    Token t;
    t.value = word;

    bool at_statement_start = _tokens.empty();

    if (!_tokens.empty())
    {
        TokenType previous = _tokens.back().type;

        if (previous == TOKEN_LBRACE || previous == TOKEN_SEMICOLON || previous == TOKEN_RBRACE)
            at_statement_start = true;
    }

    bool at_server_start = _tokens.empty() || _tokens.back().type == TOKEN_RBRACE;

    if (word == "server" && at_server_start)
        t.type = TOKEN_SERVER;
    else if (!at_statement_start)
        t.type = TOKEN_ARGUMENT;
    else if (word == "listen")
        t.type = TOKEN_LISTEN;
    else if (word == "location")
        t.type = TOKEN_LOCATION;
    else if (word == "root")
        t.type = TOKEN_ROOT;
    else if (word == "server_name")
        t.type = TOKEN_SERVER_NAME;
    else if (word == "index")
        t.type = TOKEN_INDEX;
    else if (word == "error_page")
        t.type = TOKEN_ERROR_PAGE;
    else if (word == "client_max_body_size")
        t.type = TOKEN_CLIENT_MAX_BODY_SIZE;
    else if (word == "allowed_methods")
        t.type = TOKEN_ALLOWED_METHODS;
    else if (word == "autoindex")
        t.type = TOKEN_AUTOINDEX;
    else if (word == "return")
        t.type = TOKEN_RETURN;
    else if (word == "upload")
        t.type = TOKEN_UPLOAD;
    else if (word == "upload_path")
        t.type = TOKEN_UPLOAD_PATH;
    else if (word == "cgi")
        t.type = TOKEN_CGI;
    else
        t.type = TOKEN_ARGUMENT;

    this->_tokens.push_back(t);
}


bool Tokenizer::word_to_token(std::string &line, size_t &i)
{
    std::string word = "";
    // std::cout << "\nstart Word\n";
    while (i < line.length() && 
           !std::isspace(static_cast<unsigned char>(line[i])) && 
           line[i] != '{' && line[i] != '}' && line[i] != ';' && line[i] != '#')
    {
        word += line[i];
        i++;
    }
    if (word.empty())
        return false;
    // std::cout << "Word Token: [" << word << "]" << std::endl;
    add_token_word(word);
    return true;
}

void Tokenizer::add_separator_token(std::string &line, size_t &i)
{
    Token t;
    t.value = std::string(1, line[i]);

    if(line[i] == '{')
        t.type = TOKEN_LBRACE;
    else if (line[i] == '}')
        t.type = TOKEN_RBRACE;
    else if (line[i] == ';')
        t.type = TOKEN_SEMICOLON;
    this->_tokens.push_back(t);
    i++;
}


std::vector<Token>  Tokenizer::get_token(void)
{
    return (this->_tokens);
}

void Tokenizer::str_to_token(std::string line)
{
    size_t i = 0;
    while(i < line.length())
    {
        if (std::isspace(static_cast<unsigned char>(line[i]))) 
        {
            i++;
            continue;
        }
        if (line[i] == '#') 
        {
            break;
        }
        if (line[i] == '{' || line[i] == '}' || line[i] == ';')
        {
            add_separator_token(line, i);
            continue;
        }
        word_to_token(line, i);
    }
}

    // Removes optional spaces and tabs around a header component.
    std::string trim_header_space(const std::string &value)
    {
        const std::string::size_type first = value.find_first_not_of(" \t");
        if (first == std::string::npos)
            return "";
        return value.substr(first, value.find_last_not_of(" \t") - first + 1);
    }

    // Header names and media types use case-insensitive ASCII comparisons.
    std::string lower_header_text(std::string value)
    {
        for (std::string::size_type i = 0; i < value.size(); ++i)
            if (value[i] >= 'A' && value[i] <= 'Z')
                value[i] = static_cast<char>(value[i] + ('a' - 'A'));
        return value;
    }

    // Accepts an HTTP token character for an unquoted parameter or its name.
    bool header_token_char(char c)
    {
        return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
            || (c >= '0' && c <= '9')
            || std::string("!#$%&'*+-.^_`|~").find(c) != std::string::npos;
    }

    // Reads one token or quoted value, including quoted backslash escapes.
    bool read_parameter_value(const std::string &text,
                              std::string::size_type &pos,
                              std::string &value)
    {
        value.clear();
        if (pos == text.size())
            return false;
        if (text[pos] != '"')
        {
            while (pos < text.size() && header_token_char(text[pos]))
                value += text[pos++];
            return !value.empty();
        }
        ++pos;
        while (pos < text.size())
        {
            char c = text[pos++];
            if (c == '"')
                return true;
            if (c == '\\')
            {
                if (pos == text.size())
                    return false;
                c = text[pos++];
            }
            value += c;
        }
        return false;
    }

    // MIME boundaries contain 1-70 allowed ASCII characters, no trailing space.
    bool valid_boundary(const std::string &boundary)
    {
        if (boundary.empty() || boundary.size() > 70
            || boundary[boundary.size() - 1] == ' ')
            return false;
        for (std::string::size_type i = 0; i < boundary.size(); ++i)
        {
            const char c = boundary[i];
            if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
                || (c >= '0' && c <= '9')
                || std::string("'()+_,-./:=? ").find(c) != std::string::npos))
                return false;
        }
        return true;
    }

    // Extracts a unique boundary; failure leaves boundary unchanged and sets error.
    // Inline allows standalone compilation before parse_multipart uses this helper.
    inline bool extract_boundary(const std::string &content_type,
                                 std::string &boundary,
                                 std::string &error)
    {
        error = "Invalid multipart Content-Type";
        // Reject controls before parsing, including embedded NUL and header injection.
        for (std::string::size_type i = 0; i < content_type.size(); ++i)
        {
            const unsigned char c = static_cast<unsigned char>(content_type[i]);
            if ((c < 32 && c != '\t') || c == 127)
                return false;
        }
        std::string::size_type pos = content_type.find(';');
        if (lower_header_text(trim_header_space(content_type.substr(0, pos)))
            != "multipart/form-data")
            return false;
        bool found = false;
        std::string result;
        while (pos != std::string::npos && pos < content_type.size())
        {
            ++pos;
            pos = content_type.find_first_not_of(" \t", pos);
            if (pos == std::string::npos)
                return false;
            const std::string::size_type start = pos;
            while (pos < content_type.size() && header_token_char(content_type[pos]))
                ++pos;
            const std::string name = lower_header_text(content_type.substr(start, pos - start));
            pos = content_type.find_first_not_of(" \t", pos);
            if (name.empty() || pos == std::string::npos || content_type[pos] != '=')
                return false;
            pos = content_type.find_first_not_of(" \t", pos + 1);
            if (pos == std::string::npos)
                return false;
            std::string value;
            if (!read_parameter_value(content_type, pos, value))
                return false;
            if (name == "boundary")
            {
                if (found || !valid_boundary(value))
                {
                    error = "Invalid or duplicate multipart boundary";
                    return false;
                }
                found = true;
                result = value;
            }
            pos = content_type.find_first_not_of(" \t", pos);
            if (pos != std::string::npos && content_type[pos] != ';')
                return false;
        }
        if (!found)
        {
            error = "Missing multipart boundary";
            return false;
        }
        boundary = result;
        error.clear();
        return true;
    }

    // Checks a complete delimiter line; next points past its CRLF or final EOF.
    bool read_boundary_line(const std::string &body,
                            const std::string &marker,
                            std::string::size_type pos,
                            std::string::size_type &next,
                            bool &closing)
    {
        if (body.compare(pos, marker.size(), marker) != 0)
            return false;
        pos += marker.size();
        closing = body.compare(pos, 2, "--") == 0;
        if (closing)
            pos += 2;
        // MIME delimiter lines may have optional trailing spaces or tabs.
        while (pos < body.size() && (body[pos] == ' ' || body[pos] == '\t'))
            ++pos;
        if (closing && pos == body.size())
        {
            next = pos;
            return true;
        }
        if (body.compare(pos, 2, "\r\n") != 0)
            return false;
        next = pos + 2;
        return true;
    }

    // Finds a full delimiter at a line start, skipping boundary-like file bytes.
    bool find_boundary_line(const std::string &body,
                            const std::string &marker,
                            std::string::size_type start,
                            std::string::size_type &pos,
                            std::string::size_type &next,
                            bool &closing)
    {
        pos = body.find(marker, start);
        while (pos != std::string::npos)
        {
            if ((pos == 0 || (pos >= 2 && body.compare(pos - 2, 2, "\r\n") == 0))
                && read_boundary_line(body, marker, pos, next, closing))
                return true;
            pos = body.find(marker, pos + 1);
        }
        return false;
    }

    // Splits raw parts, preserving headers and binary bytes; no files are written.
    // Preamble and epilogue are ignored. At least one part and a closing line are required.
    // On failure, parts stays unchanged. Inline permits use before public integration.
    inline bool split_multipart_parts(const std::string &body,
                                      const std::string &boundary,
                                      std::vector<std::string> &parts,
                                      std::string &error)
    {
        if (!valid_boundary(boundary))
        {
            error = "Invalid multipart boundary";
            return false;
        }
        const std::string marker = "--" + boundary;
        std::string::size_type pos;
        std::string::size_type next;
        bool closing;
        if (!find_boundary_line(body, marker, 0, pos, next, closing) || closing)
        {
            error = "Missing multipart opening boundary";
            return false;
        }
        std::vector<std::string> result;
        while (!closing)
        {
            const std::string::size_type part_start = next;
            if (!find_boundary_line(body, marker, part_start, pos, next, closing))
            {
                error = "Missing multipart closing boundary";
                return false;
            }
            // Only the CRLF immediately before a delimiter belongs to that delimiter.
            if (pos < 2 || pos - 2 < part_start)
            {
                error = "Missing CRLF before multipart boundary";
                return false;
            }
            result.push_back(body.substr(part_start, pos - 2 - part_start));
        }
        parts.swap(result);
        error.clear();
        return true;
    }

	// Rejects empty names, "." and "..", and any path separators or NUL bytes.
    bool valid_filename(const std::string &filename)
    {
        return !filename.empty()
            && filename != "." && filename != ".."
            && filename.find('/') == std::string::npos
            && filename.find('\\') == std::string::npos
            && filename.find('\0') == std::string::npos;
    }

    // Validates one unfolded part header; control bytes and invalid names are rejected.
    bool parse_part_header(const std::string &line,
                           std::string &name, std::string &value)
    {
        const std::string::size_type colon = line.find(':');
        if (colon == std::string::npos || colon == 0)
            return false;
        for (std::string::size_type i = 0; i < colon; ++i)
            if (!header_token_char(line[i]))
                return false;
        for (std::string::size_type i = colon + 1; i < line.size(); ++i)
        {
            const unsigned char c = static_cast<unsigned char>(line[i]);
            if ((c < 32 && c != '\t') || c == 127)
                return false;
        }
        name = lower_header_text(line.substr(0, colon));
        value = trim_header_space(line.substr(colon + 1));
        return true;
    }

    // Reads unique form-data parameters, reusing the quoted-value parser.
    bool parse_disposition(const std::string &text,
                           std::string &filename, bool &is_file)
    {
        std::string::size_type pos = text.find(';');
        if (lower_header_text(trim_header_space(text.substr(0, pos))) != "form-data")
            return false;
        std::map<std::string, std::string> parameters;
        while (pos != std::string::npos && pos < text.size())
        {
            pos = text.find_first_not_of(" \t", pos + 1);
            if (pos == std::string::npos)
                return false;
            const std::string::size_type start = pos;
            while (pos < text.size() && header_token_char(text[pos]))
                ++pos;
            const std::string name = lower_header_text(text.substr(start, pos - start));
            pos = text.find_first_not_of(" \t", pos);
            if (name.empty() || pos == std::string::npos || text[pos] != '=')
                return false;
            pos = text.find_first_not_of(" \t", pos + 1);
            if (pos == std::string::npos)
                return false;
            std::string value;
            if (!read_parameter_value(text, pos, value)
                || parameters.find(name) != parameters.end())
                return false;
            parameters[name] = value;
            pos = text.find_first_not_of(" \t", pos);
            if (pos != std::string::npos && text[pos] != ';')
                return false;
        }
        // Extended filename encoding is unsupported; do not silently misname a file.
        if (parameters.find("name") == parameters.end() || parameters["name"].empty()
            || parameters.find("filename*") != parameters.end())
            return false;
        is_file = parameters.find("filename") != parameters.end();
        if (is_file)
        {
            filename = parameters["filename"];
            return valid_filename(filename);
        }
        return true;
    }

    // Requires one Content-Disposition; duplicate headers and transfer encodings fail.
    bool read_part_headers(const std::string &headers,
                           std::string &filename, bool &is_file)
    {
        std::map<std::string, std::string> values;
        std::string::size_type pos = 0;
        while (pos < headers.size())
        {
            const std::string::size_type end = headers.find("\r\n", pos);
            const std::string line = headers.substr(pos,
                end == std::string::npos ? std::string::npos : end - pos);
            std::string name;
            std::string value;
            if (!parse_part_header(line, name, value)
                || values.find(name) != values.end())
                return false;
            values[name] = value;
            if (end == std::string::npos)
                break;
            pos = end + 2;
        }
        // Only raw part bytes are supported; no base64 or nested multipart decoding.
        if (values.find("content-disposition") == values.end()
            || values.find("content-transfer-encoding") != values.end())
            return false;
        if (values.find("content-type") != values.end()
            && lower_header_text(values["content-type"]).compare(0, 10, "multipart/") == 0)
            return false;
        return parse_disposition(values["content-disposition"], filename, is_file);
    }

    // Separates headers from bytes; normal form fields are validated but not returned.
    bool parse_file_part(const std::string &part, Upload::UploadedFile &file,
                         bool &is_file, std::string &error)
    {
        const std::string::size_type header_end = part.find("\r\n\r\n");
        if (header_end == std::string::npos)
        {
            error = "Missing multipart header separator";
            return false;
        }
        if (!read_part_headers(part.substr(0, header_end), file.filename, is_file))
        {
            error = "Invalid multipart headers or filename";
            return false;
        }
        // substr uses a byte count, so NUL bytes and trailing CRLF remain intact.
        if (is_file)
            file.content = part.substr(header_end + 4);
        return true;
    }

	// Checks that the directory exists, is a directory, and is writable and searchable.
    bool valid_directory(const std::string &directory)
    {
        if (directory.empty() || directory.find('\0') != std::string::npos)
        {
            errno = EINVAL;
            return false;
        }
        struct stat info;
        if (stat(directory.c_str(), &info) == -1)
            return false;
        if (!S_ISDIR(info.st_mode))
        {
            errno = ENOTDIR;
            return false;
        }
        return access(directory.c_str(), W_OK | X_OK) == 0;
    }

	// Writes the entire content to the file descriptor, handling partial writes and EINTR.
    bool write_content(int fd, const std::string &content)
    {
        std::string::size_type offset = 0;
        const std::size_t max_write = static_cast<std::size_t>(
            std::numeric_limits<ssize_t>::max());

        while (offset < content.size())
        {
            std::size_t count = content.size() - offset;
            if (count > max_write)
                count = max_write;
            ssize_t written = write(fd, content.c_str() + offset, count);
            if (written < 0)
            {
                if (errno == EINTR)
                    continue;
                return false;
            }
            if (written == 0)
            {
                errno = EIO;
                return false;
            }
            offset += static_cast<std::string::size_type>(written);
        }
        return true;
    }
}

// Saves a file to the specified directory, rejecting unsafe filenames and existing files.
// On failure, returns false and sets errno. Existing files are preserved.
bool Upload::save_file(const std::string &directory,
                       const std::string &filename,
                       const std::string &content)
{
    if (!valid_filename(filename))
    {
        errno = EINVAL;
        return false;
    }
    if (!valid_directory(directory))
        return false;

    std::string path = directory;
    if (path[path.size() - 1] != '/')
        path += '/';
    path += filename;

    int fd = open(path.c_str(), O_WRONLY | O_CREAT | O_EXCL, 0600);
    if (fd == -1)
        return false;
    if (!write_content(fd, content))
    {
        const int error = errno;
        close(fd);
        unlink(path.c_str());
        errno = error;
        return false;
    }
    if (close(fd) == -1)
    {
        const int error = errno;
        unlink(path.c_str());
        errno = error;
        return false;
    }
    return true;
}


// Parses the complete collected body; it performs no filesystem or socket operations.
// Results replace files only after every part passes validation, including form fields.
bool Upload::parse_multipart(const std::string &content_type,
                             const std::string &body,
                             std::vector<UploadedFile> &files,
                             std::string &error)
{
    std::string boundary;
    std::vector<std::string> parts;
    if (!extract_boundary(content_type, boundary, error)
        || !split_multipart_parts(body, boundary, parts, error))
        return false;
    std::vector<UploadedFile> result;
    for (std::size_t i = 0; i < parts.size(); ++i)
    {
        UploadedFile file;
        bool is_file = false;
        if (!parse_file_part(parts[i], file, is_file, error))
            return false;
        if (is_file)
            result.push_back(file);
    }
    files.swap(result);
    error.clear();
    return true;
}

bool Parser::is_valid_ipv4(const std::string& ip)
{
    int parts = 0;
    std::string::size_type start = 0;

    while(true)
    {
        std::string::size_type dot = ip.find('.', start);
        std::string part;
        int number = 0;

        if(dot == std::string::npos)
            part = ip.substr(start);
        else
            part = ip.substr(start, dot - start);

        if (part.empty() || part.size() > 3)
            return false;
        for (std::string::size_type i = 0; i < part.size(); ++i)
        {
            if (part[i] < '0' || part[i] > '9')
                return false;
            number = number * 10 + (part[i] - '0');
        }
        if (number > 255)
            return false;
        ++parts;
        if (dot == std::string::npos)
            break;
        start = dot + 1;
    }
    return parts == 4;
}


bool Parser::is_valid_port(const std::string& port)
{
    int number = 0;

    if (port.empty() || port.size() > 5)
        return false;
    for (std::string::size_type i = 0; i < port.size(); ++i)
    {
        if (port[i] < '0' || port[i] > '9')
            return false;

        number = number * 10 + (port[i] - '0');
    }
    if (number < 1 || number > 65535)
        return false;
    return true;
}

void Parser::parse_listen(ServerConfig &conf, bool &listen_defined)
{
    ++_index;

    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected address after 'listen'");

    std::string address = _tokens[_index].value;
    std::string::size_type colon = address.find(':');
    if (colon == std::string::npos || address.find(':', colon + 1) != std::string::npos)
        throw std::runtime_error("Invalid listen address: expected IPv4:port");
    std::string ip = address.substr(0, colon);
    std::string port = address.substr(colon + 1);

    if (!is_valid_ipv4(ip))
        throw std::runtime_error("Invalid IPv4 address");

    if (!is_valid_port(port))
        throw std::runtime_error("Invalid port");

    ListenConfig listen;
    listen.ip = ip;
    listen.port = std::atoi(port.c_str());

    for (std::size_t i = 0; i < conf.listens.size(); ++i)
    {
        if (conf.listens[i].ip == listen.ip && conf.listens[i].port == listen.port)
            throw std::runtime_error("Duplicate listen address");
    }
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after 'listen'");
    ++_index;
    listen_defined = true;
    conf.listens.push_back(listen);
    if (conf.listens.size() == 1)
    {
        conf.ip = listen.ip;
        conf.port = listen.port;
    }
    std::cout << "Listen IP: " << ip << std::endl;
    std::cout << "Listen port: " << port << std::endl;
}


void Parser::check_location(const ServerConfig &conf) const
{

        }
    }
}



void Parser::validate_directory(const std::string &path, const std::string &name, int permissions) const
{
    struct stat info;

    if (path.empty())
        throw std::runtime_error(name + " path is empty");
    if (stat(path.c_str(), &info) != 0)
        throw std::runtime_error(
            "Cannot access " + name + ": " + path);

    if (!S_ISDIR(info.st_mode))
        throw std::runtime_error(
            name + " is not a directory: " + path);

    if (access(path.c_str(), permissions) != 0)
        throw std::runtime_error(
            "Insufficient permissions for " + name + ": " + path);
}

void Parser::validate_executable(const std::string &path, const std::string &name) const
{
    struct stat info;

    if (path.empty())
        throw std::runtime_error(name + " path is empty");
    if (stat(path.c_str(), &info) != 0)
        throw std::runtime_error("Cannot access " + name + ": " + path);

    if (!S_ISREG(info.st_mode))
        throw std::runtime_error(name + " is not a regular file: " + path);

    if (access(path.c_str(), X_OK) != 0)
        throw std::runtime_error(name + " is not executable: " + path);
}

void Parser::parse_location_cgi(LocationConfig &location)
{

    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected CGI extension");
    const std::string ext = _tokens[_index].value;
    if (ext.size() < 2 || ext[0] != '.' || ext.find('/') != std::string::npos || ext.find('\\') != std::string::npos)
        throw std::runtime_error("Invalid CGI extension");

void Parser::parse_location_directives(LocationConfig &location)
{
    bool root_defined = false;
    bool index_defined = false;
    bool methods_defined = false;
    bool return_defined = false;
    bool autoindex_defined = false;
    bool upload_defined = false;
    bool upload_path_defined = false;
    while (_index < _tokens.size() && _tokens[_index].type != TOKEN_RBRACE)
    {
        if (_tokens[_index].type == TOKEN_ROOT)
        {
            if (root_defined)
                throw std::runtime_error("Duplicate 'root' directive in location");
            ++_index;
            parse_location_root(location);
            root_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_INDEX)
        {
            if (index_defined)
                throw std::runtime_error("Duplicate 'index' directive in location");
            ++_index;
            parse_location_index(location);
            index_defined = true; 
        }
        else if (_tokens[_index].type == TOKEN_ALLOWED_METHODS)
        {
            if(methods_defined)
                throw std::runtime_error("Duplicate 'allowed_methods' ");
            ++_index;
            parse_location_allowed_methods(location);
            methods_defined = true;
        }
        else if(_tokens[_index].type == TOKEN_RETURN)
        {
            if(return_defined)
                throw std::runtime_error("Duplicate 'return' directives in location");
            ++_index;
            parse_location_return(location);
            return_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_AUTOINDEX)
        {
            if (autoindex_defined)
                throw std::runtime_error("Duplicate 'autoindex' directive");
        
            ++_index;
            parse_location_autoindex(location);
            autoindex_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_UPLOAD)
        {
            if (upload_defined)
                throw std::runtime_error("Duplicate 'upload' directive");
            ++_index;
            parse_location_upload(location);
            upload_defined = true;
        }
        else if(_tokens[_index].type == TOKEN_UPLOAD_PATH)
        {
            if (upload_path_defined)
                throw std::runtime_error("Duplicate 'upload_path' directive");
                
            ++_index;
            parse_location_upload_path(location);
            upload_path_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_CGI)
        {
            ++_index;
            parse_location_cgi(location);
        }
        
        else
            throw std::runtime_error("Unknown directive inside location: '" + _tokens[_index].value + "'");
    }
    if (location.upload_enabled && location.upload_path.empty())
        throw std::runtime_error("'upload on' requires 'upload_path'");
}


void Parser::parse_location(ServerConfig &conf)
{
    LocationConfig location;
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected path after 'location'");
    location.path = _tokens[_index].value;
    if (location.path.empty() || location.path[0] != '/')
        throw std::runtime_error("Location path must start with '/'");
    for (std::size_t i = 0; i < conf.locations.size(); ++i)
    {
        if (conf.locations[i].path == location.path)
            throw std::runtime_error(
                "Duplicate location: " + location.path);
    }
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_LBRACE)
        throw std::runtime_error("Expected '{' after location path");
    ++_index;
    parse_location_directives(location);
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_RBRACE)
        throw std::runtime_error("Unexpected end of file: expected '}' after location");
    ++_index;
    conf.locations.push_back(location);
}

void Parser::parse_server_name(ServerConfig &conf, bool &server_name_defined)
{
    if(server_name_defined)
        throw std::runtime_error("Duplicate 'server_name' directive");
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected hostname after 'server_name'");
    conf.server_name = _tokens[_index].value;
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after 'server_name'");
    ++_index;
    server_name_defined = true;

}

void Parser::parse_error_page(ServerConfig &conf)
{
    ++_index;

    if(_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected HTTP error code after 'error_page'");
    const std::string code = _tokens[_index].value;

    if(code.size() != 3 || code[0] < '0' || code[0] > '9' || code[1] > '9' || code[1] < '0' || code[2] < '0' || code[2] > '9')
        throw std::runtime_error("Invalid HTTP error code");

    int status_code = std::atoi(code.c_str());

    if(status_code < 400 || status_code > 599)
        throw std::runtime_error("HTTP error code must be between 400 and 599");
    if(conf.error_pages.find(status_code) != conf.error_pages.end())
        throw std::runtime_error("Duplicate error_page status code");
    ++_index;
    if(_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected HTTP error page URL");

    const std::string path = _tokens[_index].value;

    if(path.empty() || path[0] != '/')
        throw std::runtime_error("Error page URI must start with '/'");
    ++_index;

    if(_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after error_page");
    ++_index;
    conf.error_pages[status_code] = path;
}


void Parser::bit_size(const std::string &suffix, std::size_t & multiplier)
{ 
    if (suffix.empty())
    multiplier = 1;
    else if (suffix == "K" || suffix == "k")
        multiplier = 1024;
    else if (suffix == "M" || suffix == "m")
        multiplier = 1024 * 1024;
    else if (suffix == "G" || suffix == "g")
        multiplier = 1024 * 1024 * 1024;
    else
        throw std::runtime_error("Invalid size suffix");
}

void Parser::get_number_multiplayer( const std::string &number_part, std::size_t &number)
{
    const std::size_t max_size = std::numeric_limits<std::size_t>::max();

    for (std::string::size_type i = 0; i < number_part.size(); ++i)
    {
        int digit = number_part[i] - '0';

        if (number > (max_size - digit) / 10)
            throw std::runtime_error("client_max_body_size overflow");

        number = number * 10 + digit;
    }
}

void Parser::parse_client_max_body_size(ServerConfig &conf, bool &defined)
{
    if (defined)
        throw std::runtime_error("Duplicate 'client_max_body_size' directive");
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected size after 'client_max_body_size'");

    const std::string value = _tokens[_index].value;
    std::string::size_type pos = 0;

    while(pos < value.size() && value[pos] >= '0'&& value[pos] <= '9')
        ++pos;
    if(pos == 0)
        throw std::runtime_error("Invalid client_max_body_size value");
    std::string number_part = value.substr(0, pos);
    std::string suffix = value.substr(pos);
    std::size_t multiplier = 1;
    std::size_t number = 0;
    bit_size(suffix, multiplier);

    get_number_multiplayer(number_part, number);

    if (number > std::numeric_limits<std::size_t>::max() / multiplier)
        throw std::runtime_error("client_max_body_size overflow");

    std::size_t size = number * multiplier;
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after client_max_body_size");
    ++_index;

    conf.client_max_body_size = size;
    defined = true;
}


void Parser::parse_location_autoindex(LocationConfig &location)
{
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected 'on' or 'off' after autoindex");
    
    const std::string value = _tokens[_index].value;

    if(value == "on")
        location.autoindex = true;
    else if (value == "off")
        location.autoindex = false;
    else
        throw std::runtime_error("Invalid autoindex value: " + value);
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after autoindex");
    ++_index;
}

void Parser::parse_server(ServerConfig &conf)
{
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SERVER)
        throw std::runtime_error("Expected 'server'");
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_LBRACE)
        throw std::runtime_error("Expected '{' after 'server'");
    ++_index;
    bool listen_defined = false;
    bool server_name_defined = false;
    bool root_defined = false;
    bool index_defined = false;
    bool client_max_body_size_defined = false;
    while (_index < _tokens.size() &&  _tokens[_index].type != TOKEN_RBRACE)
    {
        if (_tokens[_index].type == TOKEN_LISTEN)
            parse_listen(conf, listen_defined);
        else if (_tokens[_index].type == TOKEN_SERVER_NAME)
            parse_server_name(conf, server_name_defined);
        else if (_tokens[_index].type == TOKEN_ROOT)
            parse_root(conf, root_defined);
        else if (_tokens[_index].type == TOKEN_INDEX)
            parse_index(conf, index_defined);
        else if (_tokens[_index].type == TOKEN_LOCATION)
            parse_location(conf);
        else if(_tokens[_index].type == TOKEN_ERROR_PAGE)
            parse_error_page(conf);
        else if (_tokens[_index].type == TOKEN_CLIENT_MAX_BODY_SIZE)
            parse_client_max_body_size(conf, client_max_body_size_defined);
        else
            throw std::runtime_error("Unknown directive inside server");
    }
    if (_index >= _tokens.size())
        throw std::runtime_error("Unexpected end of file: expected '}'");

    ++_index;
    if (!listen_defined)
        throw std::runtime_error("Missing required directive: listen");
    if (!root_defined)
        throw std::runtime_error("Missing required directive: root");
    for (std::size_t i = 0; i < conf.locations.size(); ++i)
    {
        if (conf.locations[i].root.empty())
            conf.locations[i].root = conf.root;
    }
}


std::vector<ServerConfig> Parser::start_parse()
{
    _index = 0;
    std::vector<ServerConfig> servers;
    if (_tokens.empty())
        throw std::runtime_error("Config file is empty");


    while (_index < _tokens.size())
    {
        if (_tokens[_index].type != TOKEN_SERVER)
            throw std::runtime_error(
                "Expected 'server' at the beginning of server block"
            );

        ServerConfig conf;
        parse_server(conf);

        servers.push_back(conf);
    }
    warn_duplicate_listen_without_server_name(servers);
    return servers;
}

webserv& webserv::operator=(const webserv &ot)
{
    if(this != &ot)
        this->_config_path = ot._config_path;
    return(*this);
}

void webserv::check_path(std::string &path)
{
    struct stat info;

    if (stat(path.c_str(), &info) != 0)
        throw std::runtime_error(
            "Cannot access configuration file: " + path);

    if (!S_ISREG(info.st_mode))
        throw std::runtime_error(
            "Configuration path is not a regular file: " + path);

    if (access(path.c_str(), R_OK) != 0)
        throw std::runtime_error(
            "Configuration file is not readable: " + path);
}


void webserv::take_path()
{
    if (this->_config_path == "")
        this->_config_path = "config/default.conf";

    this->check_path(this->_config_path);
}



std::string webserv::get_path()
{
    return(this->_config_path);
}
