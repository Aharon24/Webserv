bool  HttpParser::parse_hex_size(const std::string &size_line, std::size_t &out_chunk_size)
{
    std::size_t chunk_size = 0;

    if (size_line.empty())
        return false;
    for (std::size_t j = 0; j < size_line.size(); ++j)
    {
        unsigned int digit;

        if (size_line[j] >= '0' && size_line[j] <= '9')
            digit = size_line[j] - '0';
        else if (size_line[j] >= 'a' && size_line[j] <= 'f')
            digit = size_line[j] - 'a' + 10;
        else if (size_line[j] >= 'A' && size_line[j] <= 'F')
            digit = size_line[j] - 'A' + 10;
        else
            return false;

        if (chunk_size > (static_cast<std::size_t>(-1) - digit) / 16)
            return false;

        chunk_size = chunk_size * 16 + digit;
    }

    out_chunk_size = chunk_size;
    return true;
}

ChunkedStatus  HttpParser::skip_trailers(const std::string &chunked_body, std::string::size_type &i)
{
    while (true)
    {
        std::string::size_type trailer_end = chunked_body.find("\r\n", i);

        if (trailer_end == std::string::npos)
            return CHUNKED_INCOMPLETE;

        if (trailer_end == i)
            return CHUNKED_COMPLETE;

        std::string trailer = chunked_body.substr(i, trailer_end - i);

        if (trailer.find(':') == std::string::npos)
            return CHUNKED_INVALID;

        i = trailer_end + 2;
    }
}

ChunkedStatus HttpParser::unchunk_body(const std::string &chunked_body, std::string &decoded_body, std::size_t max_body_size)
{
    decoded_body.clear();
    std::string::size_type i = 0;

    while (true)
    {
        std::string::size_type crlf = chunked_body.find("\r\n", i);

        if (crlf == std::string::npos)
            return CHUNKED_INCOMPLETE;

        std::string size_line = chunked_body.substr(i, crlf - i);

        std::string::size_type extension = size_line.find(';');

        if (extension != std::string::npos)
            size_line.erase(extension);

        if (size_line.empty())
            return CHUNKED_INVALID;

        std::size_t chunk_size = 0;
        if (!parse_hex_size(size_line, chunk_size))
            return CHUNKED_INVALID;

        i = crlf + 2;

        if (chunk_size == 0)
        {
            return skip_trailers(chunked_body, i);
        }

        if (max_body_size > 0 &&(decoded_body.size() > max_body_size || chunk_size > max_body_size - decoded_body.size()))
            return CHUNKED_TOO_LARGE;

        if (chunk_size > chunked_body.size() - i)
            return CHUNKED_INCOMPLETE;


        decoded_body.append(chunked_body, i, chunk_size);
        i += chunk_size;

        if (chunked_body.size() - i < 2)
            return CHUNKED_INCOMPLETE;

        if (chunked_body.compare(i, 2, "\r\n") != 0)
            return CHUNKED_INVALID;

        i += 2;
    }
}

void HttpParser::parse_head(const std::string &line, HttpRequest &request, bool &host_found)
{
    std::string::size_type colon = line.find(':');

    if (colon == std::string::npos || colon == 0)
        throw std::runtime_error("Invalid HTTP header");

    std::string name = line.substr(0, colon);
    std::string value = line.substr(colon + 1);

    for (std::size_t i = 0; i < name.size(); ++i)
    {
        name[i] = static_cast<char>(
            std::tolower(static_cast<unsigned char>(name[i]))
        );
    }

    while (!value.empty() && (value[0] == ' ' || value[0] == '\t'))
        value.erase(0, 1);

    while (!value.empty() &&
           (value[value.size() - 1] == ' ' ||
            value[value.size() - 1] == '\t'))
        value.erase(value.size() - 1);

    if (name == "host")
    {
        if (host_found)
            throw std::runtime_error("Duplicate Host header");

        if (value.empty())
            throw std::runtime_error("Empty Host header");
    }

    if (request.headers.find(name) != request.headers.end())
        throw std::runtime_error("Duplicate HTTP header: " + name);

    if (name == "host")
    {
        request.host = value;
        host_found = true;
    }

    request.headers.insert(
        std::map<std::string, std::string>::value_type(name, value)
    );
}

HttpRequest HttpParser::parse(const std::string &raw_request)
{
    HttpRequest request;
    std::istringstream stream(raw_request);
    std::string line;
    std::string extra;
    bool host_found = false;

    std::cout << "line ->" << raw_request << std::endl;
    if (!std::getline(stream, line))
        throw std::runtime_error("Missing request line");
    if (!line.empty() && line[line.size() - 1] == '\r')
        line.erase(line.size() - 1);
    std::istringstream request_line(line);
    if (!(request_line >> request.method >> request.path >> request.version))
        throw std::runtime_error("Invalid HTTP request line");
    
    std::string::size_type query_pos = request.path.find('?');
    if (query_pos != std::string::npos)
    {
        request.query_string = request.path.substr(query_pos + 1);
        request.path = request.path.substr(0, query_pos);
    }
    if (request.path.size() > 2048)
        throw std::runtime_error("414 URI Too Long");
    if (request_line >> extra)
        throw std::runtime_error("Invalid HTTP request line");
    if (request.version != "HTTP/1.0" && request.version != "HTTP/1.1")
        throw std::runtime_error("Unsupported HTTP version");
    if (request.path.empty() || request.path[0] != '/')
        throw std::runtime_error("Invalid request path");
    while (std::getline(stream, line))
    {
        if (!line.empty() && line[line.size() - 1] == '\r')
            line.erase(line.size() - 1);
        if (line.empty())
            break;
        if (line[0] == ' ' || line[0] == '\t')
            throw std::runtime_error("Invalid HTTP header");
        parse_head(line, request, host_found);
    }
    if (request.version == "HTTP/1.1" && !host_found)
        throw std::runtime_error("Missing Host header");
    bool has_transfer_encoding = request.headers.find("transfer-encoding") != request.headers.end();

    bool has_content_length = request.headers.find("content-length") != request.headers.end();

    if (has_transfer_encoding && has_content_length)
        throw std::runtime_error("Transfer-Encoding and Content-Length cannot be used together");
    std::string::size_type body_pos = raw_request.find("\r\n\r\n");
    if (body_pos != std::string::npos)
        request.body = raw_request.substr(body_pos + 4);
    else
    {
        body_pos = raw_request.find("\n\n");
        if (body_pos != std::string::npos)
            request.body = raw_request.substr(body_pos + 2);
    }

    std::map<std::string, std::string>::iterator it =
    request.headers.find("transfer-encoding");

    if (it != request.headers.end())
    {
        std::string transfer_encoding = it->second;

        // Make Transfer-Encoding value lowercase
        for (std::size_t i = 0; i < transfer_encoding.size(); ++i)
        {
            transfer_encoding[i] = static_cast<char>(
                std::tolower(static_cast<unsigned char>(transfer_encoding[i]))
            );
        }

        if (transfer_encoding != "chunked")
            throw std::runtime_error("Unsupported Transfer-Encoding");

        std::string decoded_body;

        ChunkedStatus status = unchunk_body(request.body, decoded_body);

        if (status == CHUNKED_INCOMPLETE)
            throw std::runtime_error("Incomplete chunked body");

        if (status == CHUNKED_INVALID)
            throw std::runtime_error("Invalid chunked body");

        if (status == CHUNKED_TOO_LARGE)
            throw std::runtime_error("Chunked body too large");

        request.body = decoded_body;
    }
    return request;
}

int init_webserv(std::string &path)
{
    webserv web(path);
    Tokenizer Tok;
    Parser pars;
    HttpParser Http_parser;
    std::vector<ServerConfig> configs;


    // pars.print_tokens();
    
    try
    {
        web.take_path();
        if (!Tok.open_file_empty_and_valid(web.get_path()))
            throw std::runtime_error("Invalid configuration file");

        Tok.create_tokens();
        pars.set_token(Tok.get_token());

        configs =  pars.start_parse();
            
        for (std::size_t i = 0; i < configs.size(); ++i)
            pars.validate_root(configs[i]);
        print_conf(configs);
        Server serv(configs);
        serv.create_socket();
        std::cout << "Socket created successfully" << std::endl;
        

        serv.bind_socket();
        serv.listen_socket();
        serv.run(Http_parser);  
    }
    catch (const std::exception& e)
    {
        std::cerr << "Config error: " << e.what() << std::endl;
        return  1;
    }
    return (0);
}

void Parser::warn_duplicate_listen_without_server_name(const std::vector<ServerConfig> &servers) const
{
    for (std::size_t i = 0; i < servers.size(); ++i)
    {
        for (std::size_t j = i + 1; j < servers.size(); ++j)
        {
            bool same_listen = false;
            std::string shared_ip;
            int shared_port = 0;

            for (std::size_t a = 0; a < servers[i].listens.size() && !same_listen; ++a)
            {
                for (std::size_t b = 0; b < servers[j].listens.size(); ++b)
                {
                    if (servers[i].listens[a].ip == servers[j].listens[b].ip && servers[i].listens[a].port == servers[j].listens[b].port)
                    {
                        same_listen = true;
                        shared_ip = servers[i].listens[a].ip;
                        shared_port = servers[i].listens[a].port;
                        break;
                    }
                }
            }
            if (same_listen && servers[i].server_name.empty() && servers[j].server_name.empty())
            {
                std::cerr << "WARNING: server blocks "
                << i + 1 << " and " << j + 1 << " use the same listen address "
                << shared_ip << ":" << shared_port << " without server_name."
                << std::endl;
            }
            else if (same_listen && !servers[i].server_name.empty() && servers[i].server_name == servers[j].server_name)
            {
                std::cerr << "WARNING: duplicate server_name '"
                << servers[i].server_name << "' in server blocks "
                << i + 1 << " and " << j + 1 << " on "
                << shared_ip << ":" << shared_port << "."
                << std::endl;
            }
        }
    }
}

bool Parser::is_valid_ipv4(const std::string& ip)
{
    int parts = 0;
    std::string::size_type start = 0;

    while(true)
    {
        std::string::size_type dot = ip.find('.', start);
        std::string part;
        int number = 0;

        if(dot == std::string::npos)
            part = ip.substr(start);
        else
            part = ip.substr(start, dot - start);

        if (part.empty() || part.size() > 3)
            return false;
        for (std::string::size_type i = 0; i < part.size(); ++i)
        {
            if (part[i] < '0' || part[i] > '9')
                return false;
            number = number * 10 + (part[i] - '0');
        }
        if (number > 255)
            return false;
        ++parts;
        if (dot == std::string::npos)
            break;
        start = dot + 1;
    }
    return parts == 4;
}


bool Parser::is_valid_port(const std::string& port)
{
    int number = 0;

    if (port.empty() || port.size() > 5)
        return false;
    for (std::string::size_type i = 0; i < port.size(); ++i)
    {
        if (port[i] < '0' || port[i] > '9')
            return false;

        number = number * 10 + (port[i] - '0');
    }
    if (number < 1 || number > 65535)
        return false;
    return true;
}

void Parser::parse_listen(ServerConfig &conf, bool &listen_defined)
{
    ++_index;

    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected address after 'listen'");

    std::string address = _tokens[_index].value;
    std::string::size_type colon = address.find(':');
    if (colon == std::string::npos || address.find(':', colon + 1) != std::string::npos)
        throw std::runtime_error("Invalid listen address: expected IPv4:port");
    std::string ip = address.substr(0, colon);
    std::string port = address.substr(colon + 1);

    if (!is_valid_ipv4(ip))
        throw std::runtime_error("Invalid IPv4 address");

    if (!is_valid_port(port))
        throw std::runtime_error("Invalid port");

    ListenConfig listen;
    listen.ip = ip;
    listen.port = std::atoi(port.c_str());

    for (std::size_t i = 0; i < conf.listens.size(); ++i)
    {
        if (conf.listens[i].ip == listen.ip && conf.listens[i].port == listen.port)
            throw std::runtime_error("Duplicate listen address");
    }
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after 'listen'");
    ++_index;
    listen_defined = true;
    conf.listens.push_back(listen);
    if (conf.listens.size() == 1)
    {
        conf.ip = listen.ip;
        conf.port = listen.port;
    }
    std::cout << "Listen IP: " << ip << std::endl;
    std::cout << "Listen port: " << port << std::endl;
}


void Parser::check_location(const ServerConfig &conf) const
{
    for(size_t i = 0; i < conf.locations.size(); ++i)
    {
        const LocationConfig &location = conf.locations[i];

        if (location.root.empty() && location.return_code == 0)
            throw std::runtime_error("Missing root for location: " + location.path);


        if (!location.root.empty())
            validate_directory(location.root, "location root", R_OK | X_OK);
        // Check upload directory
        if (location.upload_enabled)
            validate_directory(location.upload_path, "upload path",W_OK | X_OK);
        for (std::size_t j = 0; j < location.cgi.size(); ++j)
        {
            validate_executable(location.cgi[j].interpreter, "CGI interpreter");
        }
    }
}



void Parser::validate_directory(const std::string &path, const std::string &name, int permissions) const
{
    struct stat info;

    if (path.empty())
        throw std::runtime_error(name + " path is empty");
    if (stat(path.c_str(), &info) != 0)
        throw std::runtime_error(
            "Cannot access " + name + ": " + path);

    if (!S_ISDIR(info.st_mode))
        throw std::runtime_error(
            name + " is not a directory: " + path);

    if (access(path.c_str(), permissions) != 0)
        throw std::runtime_error(
            "Insufficient permissions for " + name + ": " + path);
}

void Parser::validate_executable(const std::string &path, const std::string &name) const
{
    struct stat info;

    if (path.empty())
        throw std::runtime_error(name + " path is empty");
    if (stat(path.c_str(), &info) != 0)
        throw std::runtime_error("Cannot access " + name + ": " + path);

    if (!S_ISREG(info.st_mode))
        throw std::runtime_error(name + " is not a regular file: " + path);

    if (access(path.c_str(), X_OK) != 0)
        throw std::runtime_error(name + " is not executable: " + path);
}

void Parser::parse_location_cgi(LocationConfig &location)

void Parser::parse_location_directives(LocationConfig &location)
{
    bool root_defined = false;
    bool index_defined = false;
    bool methods_defined = false;
    bool return_defined = false;
    bool autoindex_defined = false;
    bool upload_defined = false;
    bool upload_path_defined = false;
    while (_index < _tokens.size() && _tokens[_index].type != TOKEN_RBRACE)
    {
        if (_tokens[_index].type == TOKEN_ROOT)
        {
            if (root_defined)
                throw std::runtime_error("Duplicate 'root' directive in location");
            ++_index;
            parse_location_root(location);
            root_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_INDEX)
        {
            if (index_defined)
                throw std::runtime_error("Duplicate 'index' directive in location");
            ++_index;
            parse_location_index(location);
            index_defined = true; 
        }
        else if (_tokens[_index].type == TOKEN_ALLOWED_METHODS)
        {
            if(methods_defined)
                throw std::runtime_error("Duplicate 'allowed_methods' ");
            ++_index;
            parse_location_allowed_methods(location);
            methods_defined = true;
        }
        else if(_tokens[_index].type == TOKEN_RETURN)
        {
            if(return_defined)
                throw std::runtime_error("Duplicate 'return' directives in location");
            ++_index;
            parse_location_return(location);
            return_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_AUTOINDEX)
        {
            if (autoindex_defined)
                throw std::runtime_error("Duplicate 'autoindex' directive");
        
            ++_index;
            parse_location_autoindex(location);
            autoindex_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_UPLOAD)
        {
            if (upload_defined)
                throw std::runtime_error("Duplicate 'upload' directive");
            ++_index;
            parse_location_upload(location);
            upload_defined = true;
        }
        else if(_tokens[_index].type == TOKEN_UPLOAD_PATH)
        {
            if (upload_path_defined)
                throw std::runtime_error("Duplicate 'upload_path' directive");
                
            ++_index;
            parse_location_upload_path(location);
            upload_path_defined = true;
        }
        else if (_tokens[_index].type == TOKEN_CGI)
        {
            ++_index;
            parse_location_cgi(location);
        }
        
        else
            throw std::runtime_error("Unknown directive inside location: '" + _tokens[_index].value + "'");
    }
    if (location.upload_enabled && location.upload_path.empty())
        throw std::runtime_error("'upload on' requires 'upload_path'");
}


void Parser::parse_location(ServerConfig &conf)
{
    LocationConfig location;
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected path after 'location'");
    location.path = _tokens[_index].value;
    if (location.path.empty() || location.path[0] != '/')
        throw std::runtime_error("Location path must start with '/'");
    for (std::size_t i = 0; i < conf.locations.size(); ++i)
    {
        if (conf.locations[i].path == location.path)
            throw std::runtime_error(
                "Duplicate location: " + location.path);
    }
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_LBRACE)
        throw std::runtime_error("Expected '{' after location path");
    ++_index;
    parse_location_directives(location);
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_RBRACE)
        throw std::runtime_error("Unexpected end of file: expected '}' after location");
    ++_index;
    conf.locations.push_back(location);
}

void Parser::parse_error_page(ServerConfig &conf)
{
    ++_index;

    if(_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected HTTP error code after 'error_page'");
    const std::string code = _tokens[_index].value;

    if(code.size() != 3 || code[0] < '0' || code[0] > '9' || code[1] > '9' || code[1] < '0' || code[2] < '0' || code[2] > '9')
        throw std::runtime_error("Invalid HTTP error code");

    int status_code = std::atoi(code.c_str());

    if(status_code < 400 || status_code > 599)
        throw std::runtime_error("HTTP error code must be between 400 and 599");
    if(conf.error_pages.find(status_code) != conf.error_pages.end())
        throw std::runtime_error("Duplicate error_page status code");
    ++_index;
    if(_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected HTTP error page URL");

    const std::string path = _tokens[_index].value;

    if(path.empty() || path[0] != '/')
        throw std::runtime_error("Error page URI must start with '/'");
    ++_index;

    if(_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after error_page");
    ++_index;
    conf.error_pages[status_code] = path;
}


void Parser::bit_size(const std::string &suffix, std::size_t & multiplier)
{ 
    if (suffix.empty())
    multiplier = 1;
    else if (suffix == "K" || suffix == "k")
        multiplier = 1024;
    else if (suffix == "M" || suffix == "m")
        multiplier = 1024 * 1024;
    else if (suffix == "G" || suffix == "g")
        multiplier = 1024 * 1024 * 1024;
    else
        throw std::runtime_error("Invalid size suffix");
}

void Parser::get_number_multiplayer( const std::string &number_part, std::size_t &number)
{
    const std::size_t max_size = std::numeric_limits<std::size_t>::max();

    for (std::string::size_type i = 0; i < number_part.size(); ++i)
    {
        int digit = number_part[i] - '0';

        if (number > (max_size - digit) / 10)
            throw std::runtime_error("client_max_body_size overflow");

        number = number * 10 + digit;
    }
}

void Parser::parse_client_max_body_size(ServerConfig &conf, bool &defined)
{
    if (defined)
        throw std::runtime_error("Duplicate 'client_max_body_size' directive");
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_ARGUMENT)
        throw std::runtime_error("Expected size after 'client_max_body_size'");

    const std::string value = _tokens[_index].value;
    std::string::size_type pos = 0;

    while(pos < value.size() && value[pos] >= '0'&& value[pos] <= '9')
        ++pos;
    if(pos == 0)
        throw std::runtime_error("Invalid client_max_body_size value");
    std::string number_part = value.substr(0, pos);
    std::string suffix = value.substr(pos);
    std::size_t multiplier = 1;
    std::size_t number = 0;
    bit_size(suffix, multiplier);

    get_number_multiplayer(number_part, number);

    if (number > std::numeric_limits<std::size_t>::max() / multiplier)
        throw std::runtime_error("client_max_body_size overflow");

    std::size_t size = number * multiplier;
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SEMICOLON)
        throw std::runtime_error("Expected ';' after client_max_body_size");
    ++_index;

    conf.client_max_body_size = size;
    defined = true;
}

void Parser::parse_server(ServerConfig &conf)
{
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_SERVER)
        throw std::runtime_error("Expected 'server'");
    ++_index;
    if (_index >= _tokens.size() || _tokens[_index].type != TOKEN_LBRACE)
        throw std::runtime_error("Expected '{' after 'server'");
    ++_index;
    bool listen_defined = false;
    bool server_name_defined = false;
    bool root_defined = false;
    bool index_defined = false;
    bool client_max_body_size_defined = false;
    while (_index < _tokens.size() &&  _tokens[_index].type != TOKEN_RBRACE)
    {
        if (_tokens[_index].type == TOKEN_LISTEN)
            parse_listen(conf, listen_defined);
        else if (_tokens[_index].type == TOKEN_SERVER_NAME)
            parse_server_name(conf, server_name_defined);
        else if (_tokens[_index].type == TOKEN_ROOT)
            parse_root(conf, root_defined);
        else if (_tokens[_index].type == TOKEN_INDEX)
            parse_index(conf, index_defined);
        else if (_tokens[_index].type == TOKEN_LOCATION)
            parse_location(conf);
        else if(_tokens[_index].type == TOKEN_ERROR_PAGE)
            parse_error_page(conf);
        else if (_tokens[_index].type == TOKEN_CLIENT_MAX_BODY_SIZE)
            parse_client_max_body_size(conf, client_max_body_size_defined);
        else
            throw std::runtime_error("Unknown directive inside server");
    }
    if (_index >= _tokens.size())
        throw std::runtime_error("Unexpected end of file: expected '}'");

    ++_index;
    if (!listen_defined)
        throw std::runtime_error("Missing required directive: listen");
    if (!root_defined)
        throw std::runtime_error("Missing required directive: root");
    for (std::size_t i = 0; i < conf.locations.size(); ++i)
    {
        if (conf.locations[i].root.empty())
            conf.locations[i].root = conf.root;
    }
}

void Server::set_nonblocking(int fd)
{
    int flags = fcntl(fd, F_GETFL, 0);

    if (flags == -1)
        throw std::runtime_error("Failed to get socket flags");

    if (fcntl(fd, F_SETFL, flags | O_NONBLOCK) == -1)
        throw std::runtime_error("Failed to set non-blocking mode");
}

void Server::create_socket()
{
    for (std::size_t i = 0; i < _listens.size(); ++i)
    {
        int server_fd = socket(AF_INET, SOCK_STREAM, 0);
        if (server_fd == -1)
            throw std::runtime_error("Failed to create socket");
        int reuse = 1;
        if (setsockopt(server_fd, SOL_SOCKET, SO_REUSEADDR, &reuse, sizeof(reuse)) == -1)
        {
            close(server_fd);
            throw std::runtime_error("Failed to set SO_REUSEADDR");
        }
        try
        {
            set_nonblocking(server_fd);
        }
        catch (...)
        {
            close(server_fd);
            server_fd = -1;
            throw;
        }
        _server_fds.push_back(server_fd);
        std::cout << "Created socket FD: " << server_fd << " for port: " << _listens[i].port << std::endl;
    }

}


void Server::listen_socket()
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        if (listen(_server_fds[i], 128) == -1)
            throw std::runtime_error("Failed to listen on socket");
    }
}

void Server::bind_socket()
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        struct sockaddr_in address;

        std::memset(&address, 0, sizeof(address));
        address.sin_family = AF_INET;
        address.sin_port = htons(_listens[i].port);
        if (inet_pton(AF_INET,_listens[i].ip.c_str(), &address.sin_addr) != 1)
            throw std::runtime_error("Invalid IPv4 address");
        if (bind(_server_fds[i], reinterpret_cast<struct sockaddr *>(&address), sizeof(address)) == -1)
            throw std::runtime_error("Failed to bind socket");
        std::cout << "Bind successful: " << _listens[i].ip << ":" << _listens[i].port << std::endl;
    }
}

int Server::accept_connection(int server_fd)
{
    int client_fd = accept(server_fd, NULL, NULL);

    if (client_fd == -1)
    {
        if (errno == EINTR || errno == EAGAIN || errno == EWOULDBLOCK || errno == ECONNABORTED)
            return -1;
        throw std::runtime_error("Failed to accept connection");
    }
    try
    {
        set_nonblocking(client_fd);
    }
    catch (...)
    {
        close(client_fd);
        return -1;
    }
    return client_fd;
}

const LocationConfig* Server::find_location(const std::string &path, const ServerConfig &conf) const
{
    const LocationConfig *best = NULL;

    for (std::size_t i = 0; i < conf.locations.size(); ++i)
    {
        const std::string &prefix = conf.locations[i].path;
        bool matches = false;

        if (prefix.empty())
            continue;

        if (path.size() >= prefix.size() && path.compare(0, prefix.size(), prefix) == 0)
        {
            if (path.size() == prefix.size())
                matches = true;
            else if (prefix == "/")
                matches = true;
            else if (path[prefix.size()] == '/')
                matches = true;
        }
        if (matches &&(best == NULL || prefix.size() > best->path.size()))
            best = &conf.locations[i];
    }
    return best;
}

    if (stat(file_path.c_str(), &info) == 0 && S_ISDIR(info.st_mode))
    {
        std::string index;
    
        if (location != NULL && !location->index.empty())
            index = location->index;
        else
            index = conf.index;
        if (!file_path.empty() && file_path[file_path.size() - 1] != '/')
            file_path += '/';
        file_path += index;

    if(!file)
    {
        status = "404 Not Found";
        body = "404 Not Found\n";
    }
    else
    {
        std::ostringstream content;
        content << file.rdbuf();
        body = content.str();
        status = "200 OK";
    }
}

void Server::handle_delete(const HttpRequest &request, const ServerConfig &conf,std::string &status, std::string &body)
{
    const LocationConfig *location = find_location(request.path, conf);

    if (location != NULL && !location->allowed_methods.empty())
    {
        bool allowed = false;

        for (std::size_t i = 0; i < location->allowed_methods.size(); ++i)
        {
            if (location->allowed_methods[i] == "DELETE")
                allowed = true;
        }
        if (!allowed)
        {
            status = "405 Method Not Allowed";
            body = "405 Method Not Allowed\n";
            return;
        }
    }

    if (request.path.empty() || request.path[0] != '/'
        || request.path.find("..") != std::string::npos)
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    std::string root = conf.root;
    if (location != NULL && !location->root.empty())
        root = location->root;

    if (root.empty())
    {
        status = "500 Internal Server Error";
        body = "Invalid server root\n";
        return;
    }

    const std::string file_path = build_file_path(request, location, conf);

    std::string normalized_root = root;
    if (normalized_root[normalized_root.size() - 1] != '/')
        normalized_root += '/';

    if (file_path.compare(0, normalized_root.size(), normalized_root) != 0)
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    struct stat info;
    if (stat(file_path.c_str(), &info) == -1)
    {
        if (errno == ENOENT || errno == ENOTDIR)
            status = "404 Not Found";
        else if (errno == EACCES || errno == EPERM)
            status = "403 Forbidden";
        else
            status = "500 Internal Server Error";
        body = status + "\n";
        return;
    }

    if (!S_ISREG(info.st_mode))
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
        return;
    }

    if (unlink(file_path.c_str()) == -1)
    {
        if (errno == ENOENT || errno == ENOTDIR)
            status = "404 Not Found";
        else if (errno == EACCES || errno == EPERM || errno == EROFS)
            status = "403 Forbidden";
        else
            status = "500 Internal Server Error";

        body = status + "\n";
        return;
    }

    status = "204 No Content";
    body.clear();
}

std::string Server::build_response(const HttpRequest &request,const ServerConfig &conf)
{
    std::string body;
    std::string status;

    const LocationConfig *location = find_location(request.path, conf);

    if (location != NULL && location->return_code != 0)
    {
        std::ostringstream ss;
        ss << request.version << " " << location->return_code << " Moved Permanently\r\n"
           << "Location: " << location->return_url << "\r\n"
           << "Content-Length: 0\r\n"
           << "Connection: close\r\n\r\n";
        return ss.str();
    }

    if (request.path.empty() ||
        request.path[0] != '/' ||
        request.path.find("..") != std::string::npos)
    {
        status = "403 Forbidden";
        body = "403 Forbidden\n";
    }
    else if (request.method == "GET")
        handle_get(request, conf, status, body);
    else if (request.method == "POST")
        handle_post(request, status, body);
    else if (request.method == "DELETE")
        handle_delete(request, conf, status, body);
    else
    {
        status = "405 Method Not Allowed";
        body = "405 Method Not Allowed\n";
    }

    return create_response_data(request, status, body) + body;
}



std::string Server::build_file_path(const HttpRequest &request,const LocationConfig *location,const ServerConfig &conf) const
{
    std::string root;
    std::string relative_path;

    if (location != NULL)
    {
        if (location->root.empty())
            root = conf.root;
        else
            root = location->root;
        relative_path = request.path.substr(location->path.size());
    }
    else
    {
        root = conf.root;
        relative_path = request.path;   
    }
    if (!relative_path.empty() && relative_path[0] == '/')
        relative_path.erase(0, 1);
    if (!root.empty() && root[root.size() - 1] != '/')
        root += '/';
    return root + relative_path;
}

bool Server::get_content_length(const std::string &headers_part,std::size_t &content_length) const
{
    content_length = 0;
    bool found = false;
    std::string::size_type line_start = 0;

    while (line_start < headers_part.size())
    {
        std::string::size_type line_end = headers_part.find("\r\n", line_start);

        if (line_end == std::string::npos)
            line_end = headers_part.size();

        std::string::size_type colon = headers_part.find(':', line_start);

        if (colon != std::string::npos && colon < line_end)
        {
            std::string name = headers_part.substr(line_start, colon - line_start);

            for (std::string::size_type i = 0; i < name.size(); ++i)
            {
                if (name[i] >= 'A' && name[i] <= 'Z')
                    name[i] = name[i] - 'A' + 'a';
            }

            if (name == "content-length")
            {
                if (found)
                    return false;

                found = true;

                std::string::size_type start = colon + 1;

                while (start < line_end &&
                       (headers_part[start] == ' ' ||
                        headers_part[start] == '\t'))
                    ++start;

                std::string::size_type end = line_end;

                while (end > start && (headers_part[end - 1] == ' ' || headers_part[end - 1] == '\t'))
                    --end;

                if (start == end)
                    return false;

                std::size_t value = 0;

                for (std::string::size_type i = start; i < end; ++i)
                {
                    if (headers_part[i] < '0' ||
                        headers_part[i] > '9')
                        return false;

                    std::size_t digit =
                        static_cast<std::size_t>(headers_part[i] - '0');

                    if (value >
                        (std::numeric_limits<std::size_t>::max() - digit) / 10)
                        return false;

                    value = value * 10 + digit;
                }

                content_length = value;
            }
        }

        if (line_end == headers_part.size())
            break;

        line_start = line_end + 2;
    }

    return true;
}

bool Server::read_client_data(ClientConnection &client, bool &stop_processing)
{
    const std::size_t MAX_HEADER_SIZE = 32 * 1024;

    char buffer[4096];
    stop_processing = false;

    ssize_t bytes_read = recv(client.fd, buffer, sizeof(buffer), 0);

    if (bytes_read == 0)
        return false;

    if (bytes_read < 0)
    {
        stop_processing = true;
        return true;
    }

    client.last_activity = time(NULL);

    client.request_buffer.append(
        buffer,
        static_cast<std::string::size_type>(bytes_read)
    );

    std::string::size_type header_end = client.request_buffer.find("\r\n\r\n");

    if (header_end == std::string::npos)
    {
        if (client.request_buffer.size() > MAX_HEADER_SIZE)
        {
            client.header_too_large = true;
            stop_processing = true;
        }
    }
    else if (header_end + 4 > MAX_HEADER_SIZE)
    {
        client.header_too_large = true;
        stop_processing = true;
    }

    return true;
}

bool Server::is_chunked(const std::string &headers_part) const
{
    std::string::size_type line_start = 0;

    while (line_start < headers_part.size())
    {
        std::string::size_type line_end = headers_part.find("\r\n", line_start);

        if (line_end == std::string::npos)
            line_end = headers_part.size();

        std::string::size_type colon = headers_part.find(':', line_start);

        if (colon != std::string::npos && colon < line_end)
        {
            std::string name = headers_part.substr(line_start, colon - line_start);

            for (std::string::size_type i = 0; i < name.size(); ++i)
            {
                if (name[i] >= 'A' && name[i] <= 'Z')
                    name[i] = name[i] - 'A' + 'a';
            }

            if (name == "transfer-encoding")
            {
                std::string::size_type token_start = colon + 1;

                while (token_start < line_end)
                {
                    std::string::size_type comma_pos =
                        headers_part.find(',', token_start);

                    if (comma_pos == std::string::npos || comma_pos > line_end)
                        comma_pos = line_end;

                    std::string::size_type token_end = comma_pos;

                    while (token_start < token_end && (headers_part[token_start] == ' ' || headers_part[token_start] == '\t'))
                        ++token_start;

                    while (token_end > token_start &&
                           (headers_part[token_end - 1] == ' ' ||
                            headers_part[token_end - 1] == '\t'))
                        --token_end;

                    std::string token =
                        headers_part.substr(
                            token_start, token_end - token_start);

                    for (std::string::size_type i = 0;
                         i < token.size(); ++i)
                    {
                        if (token[i] >= 'A' && token[i] <= 'Z')
                            token[i] = token[i] - 'A' + 'a';
                    }

                    if (token == "chunked")
                        return true;

                    if (comma_pos == line_end)
                        break;

                    token_start = comma_pos + 1;
                }
            }
        }

        if (line_end == headers_part.size())
            break;

        line_start = line_end + 2;
    }

    return false;
}

{

    dummy_req.version = "HTTP/1.1";

    std::string status = "431 Request Header Fields Too Large";

bool Server::handle_client_event(ClientConnection &client, HttpParser &http_parser)
{
    bool stop_processing = false;
    if (!read_client_data(client, stop_processing))
        return false;
    if (client.header_too_large)
    {
        handle_large_header(client);
        return true;
    }
    if (stop_processing)
        return true;

    std::string::size_type header_end = client.request_buffer.find("\r\n\r\n");
    if (header_end == std::string::npos)
        return true; 
    const ServerConfig &conf = get_server_config(client);

    std::string headers_part = client.request_buffer.substr(0, header_end);
    std::string::size_type body_start = header_end + 4;

    if (is_chunked(headers_part))
    {
        std::string decoded_body;

        ChunkedStatus chunked_status = HttpParser::unchunk_body(client.request_buffer.substr(body_start), decoded_body, conf.client_max_body_size);

        if (chunked_status == CHUNKED_INCOMPLETE)
            return true;

        if (chunked_status == CHUNKED_INVALID)
        {
            parse_and_respond(client, http_parser, conf);
            return true;
        }
        if (chunked_status == CHUNKED_TOO_LARGE)
        {
            HttpRequest dummy_req;
            dummy_req.version = "HTTP/1.1";

            std::string status = "413 Payload Too Large";
            std::string body = status + "\n";

            client.response_buffer =
                create_response_data(dummy_req, status, body) + body;

            client.request_complete = true;
            client.response_ready = true;
            return true;
        }

        if (check_max_body_size(client, decoded_body.size(), conf))
            return true;
    }
    else
    {
        std::size_t content_length = 0;

        if (!get_content_length(headers_part, content_length))
        {
            HttpRequest dummy_req;
            dummy_req.version = "HTTP/1.1";

            std::string status = "400 Bad Request";
            std::string body = status + "\n";

            client.response_buffer =
                create_response_data(dummy_req, status, body) + body;

            client.request_complete = true;
            client.response_ready = true;
            return true;
        }

        if (check_max_body_size(client, content_length, conf))
            return true;

        if (client.request_buffer.size() - body_start < content_length)
            return true;
    }

    parse_and_respond(client, http_parser, conf);
    return true;
}

const ServerConfig &Server::get_server_config(const ClientConnection &client) const

    if (client.listener_index >= _listen_config_indices.size())

    std::size_t config_index = _listen_config_indices[client.listener_index];

}



bool Server::handle_client_write(ClientConnection &client)
{
    if (client.sent_bytes >= client.response_buffer.size())
        return false;

    ssize_t bytes_sent = send(client.fd, client.response_buffer.c_str() + client.sent_bytes,
        client.response_buffer.size() - client.sent_bytes,
        0);

    if (bytes_sent < 0)
        return false;

    if (bytes_sent == 0)
        return false;

    client.sent_bytes += static_cast<std::size_t>(bytes_sent);
    client.last_activity = time(NULL);

    if (client.sent_bytes < client.response_buffer.size())
        return true;
    std::cout << "Response fully sent to client FD: " << client.fd << std::endl;

    bool close_connection =(client.response_buffer.find("Connection: close") != std::string::npos);
    if (close_connection)
        return false;
    client.reset();
    return true;
}  

const ServerConfig *Server::find_server_config(const std::string &host) const
{
    if (_configs.empty())
        return NULL;
    
    std::string host_name = host;

    std::string::size_type colon = host_name.find(':');
    if (colon != std::string::npos)
        host_name.erase(colon);
    for (std::size_t i = 0; i < host_name.size(); ++i)
    {
        host_name[i] = static_cast<char>(std::tolower(static_cast<unsigned char>(host_name[i])));
    }
    for (std::size_t i = 0; i < _configs.size(); ++i)
    {
        if (_configs[i].server_name.empty())
            continue;

        std::string configured_name = _configs[i].server_name;

        for (std::size_t j = 0; j < configured_name.size(); ++j)
        {
            configured_name[j] = static_cast<char>(std::tolower(static_cast<unsigned char>(configured_name[j])));
        }
        if (configured_name == host_name)
            return &_configs[i];
    }
    return  NULL;
}

std::vector<struct pollfd> Server::build_poll_fds(const std::vector<ClientConnection> &clients) const
{
    std::vector<struct pollfd> fds;

    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        struct pollfd server_poll;
        server_poll.fd = _server_fds[i];
        server_poll.events = POLLIN;
        server_poll.revents = 0;
        fds.push_back(server_poll);
    }

    for (std::size_t i = 0; i < clients.size(); ++i)
    {
        struct pollfd client_poll;
        client_poll.fd = clients[i].fd;
        client_poll.events = clients[i].response_ready ? POLLOUT : POLLIN;
        client_poll.revents = 0;
        fds.push_back(client_poll);
    }
    return fds;
}


void Server::accept_ready_connections(const std::vector<struct pollfd> &fds, std::vector<ClientConnection> &clients)
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        if (_server_fds[i] == -1)
            continue;

        if (fds[i].revents & POLLIN)
        {
            int client_fd = accept_connection(_server_fds[i]);

            if (client_fd != -1)
            {
                std::cout << "Accepted client: " << client_fd << std::endl;
                clients.push_back(ClientConnection(client_fd, i));
            }
        }

        if (fds[i].revents & (POLLERR | POLLHUP | POLLNVAL))
        {
            std::cerr << "Error: Listening socket FD " << _server_fds[i]
                      << " encountered error/hangup/invalid state." << std::endl;
            close(_server_fds[i]);
            _server_fds[i] = -1;
        }
    }
}


int Server::wait_for_events(std::vector<struct pollfd> &fds, int timeout_ms) const
{
    if (fds.empty())
        return 0;

    int result = poll(&fds[0], fds.size(),timeout_ms);

    if (result == -1)
    {
        if (errno == EINTR)
            return -1;

        throw std::runtime_error("poll failed");
    }
    return result;
}

void Server::process_client_events(const std::vector<struct pollfd> &fds, std::vector<ClientConnection> &clients, HttpParser &parser)
{
    for (std::size_t i = _server_fds.size(); i < fds.size(); ++i)
    {
        int client_fd = fds[i].fd;
        short events = fds[i].revents;
        if (events == 0)
            continue;
        std::vector<ClientConnection>::iterator it = clients.begin();

        while (it != clients.end() && it->fd != client_fd)
            ++it;

        if (it == clients.end())
            continue;

        if (events & POLLNVAL)
        {
            remove_client(client_fd, clients);
            continue;
        }
        bool keep_client = true;
        if (events & POLLIN)
            keep_client = handle_client_event(*it, parser);

        if (keep_client && (events & POLLOUT))
            keep_client = handle_client_write(*it);

        if (events & POLLERR)
            keep_client = false;

        if ((events & POLLHUP) && !(events & POLLIN) && !(events & POLLOUT))
            keep_client = false;

        if (!keep_client)
            remove_client(client_fd, clients);
    }
}


void Server::check_timeouts(std::vector<ClientConnection> &clients, time_t timeout_seconds)
{
    time_t current_time = time(NULL);
    for(std::size_t i = 0; i < clients.size(); )
    {
        if(current_time - clients[i].last_activity > timeout_seconds)
        {
            std::cout << "Client FD " << clients[i].fd << " timed out. Closing connection." << std::endl;
            close(clients[i].fd);
            clients.erase(clients.begin() + i);
        }
        else
            ++i;
    }
}

void Server::run(HttpParser &parser)
{
    std::vector<ClientConnection> clients;
    const time_t TIMEOUT_SECONDS = 5;
    while (true)
    {
        check_timeouts(clients, TIMEOUT_SECONDS);

        std::vector<struct pollfd> fds;

        fds = build_poll_fds(clients);

        int result = wait_for_events(fds, 1000);
        if (result <= 0)
            continue;       
        accept_ready_connections(fds, clients);
        
        process_client_events(fds, clients, parser);
    }
}


Server::~Server()
{
    for (std::size_t i = 0; i < _server_fds.size(); ++i)
    {
        if (_server_fds[i] != -1)
            close(_server_fds[i]);
    }
}

void Tokenizer::add_token_word(std::string &word)
{
    Token t;
    t.value = word;

    bool at_statement_start = _tokens.empty();

    if (!_tokens.empty())
    {
        TokenType previous = _tokens.back().type;

        if (previous == TOKEN_LBRACE || previous == TOKEN_SEMICOLON || previous == TOKEN_RBRACE)
            at_statement_start = true;
    }

    bool at_server_start = _tokens.empty() || _tokens.back().type == TOKEN_RBRACE;

    if (word == "server" && at_server_start)
        t.type = TOKEN_SERVER;
    else if (!at_statement_start)
        t.type = TOKEN_ARGUMENT;
    else if (word == "listen")
        t.type = TOKEN_LISTEN;
    else if (word == "location")
        t.type = TOKEN_LOCATION;
    else if (word == "root")
        t.type = TOKEN_ROOT;
    else if (word == "server_name")
        t.type = TOKEN_SERVER_NAME;
    else if (word == "index")
        t.type = TOKEN_INDEX;
    else if (word == "error_page")
        t.type = TOKEN_ERROR_PAGE;
    else if (word == "client_max_body_size")
        t.type = TOKEN_CLIENT_MAX_BODY_SIZE;
    else if (word == "allowed_methods")
        t.type = TOKEN_ALLOWED_METHODS;
    else if (word == "autoindex")
        t.type = TOKEN_AUTOINDEX;
    else if (word == "return")
        t.type = TOKEN_RETURN;
    else if (word == "upload")
        t.type = TOKEN_UPLOAD;
    else if (word == "upload_path")
        t.type = TOKEN_UPLOAD_PATH;
    else if (word == "cgi")
        t.type = TOKEN_CGI;
    else
        t.type = TOKEN_ARGUMENT;

    this->_tokens.push_back(t);
}


bool Tokenizer::word_to_token(std::string &line, size_t &i)
{
    std::string word = "";
    // std::cout << "\nstart Word\n";
    while (i < line.length() && 
           !std::isspace(static_cast<unsigned char>(line[i])) && 
           line[i] != '{' && line[i] != '}' && line[i] != ';' && line[i] != '#')
    {
        word += line[i];
        i++;
    }
    if (word.empty())
        return false;
    // std::cout << "Word Token: [" << word << "]" << std::endl;
    add_token_word(word);
    return true;
}

void Tokenizer::add_separator_token(std::string &line, size_t &i)
{
    Token t;
    t.value = std::string(1, line[i]);

    if(line[i] == '{')
        t.type = TOKEN_LBRACE;
    else if (line[i] == '}')
        t.type = TOKEN_RBRACE;
    else if (line[i] == ';')
        t.type = TOKEN_SEMICOLON;
    this->_tokens.push_back(t);
    i++;
}

void Tokenizer::str_to_token(std::string line)
{
    size_t i = 0;
    while(i < line.length())
    {
        if (std::isspace(static_cast<unsigned char>(line[i]))) 
        {
            i++;
            continue;
        }
        if (line[i] == '#') 
        {
            break;
        }
        if (line[i] == '{' || line[i] == '}' || line[i] == ';')
        {
            add_separator_token(line, i);
            continue;
        }
        word_to_token(line, i);
    }
}

    // Removes optional spaces and tabs around a header component.
    std::string trim_header_space(const std::string &value)
    {
        const std::string::size_type first = value.find_first_not_of(" \t");
        if (first == std::string::npos)
            return "";
        return value.substr(first, value.find_last_not_of(" \t") - first + 1);
    }

    // Header names and media types use case-insensitive ASCII comparisons.
    std::string lower_header_text(std::string value)
    {
        for (std::string::size_type i = 0; i < value.size(); ++i)
            if (value[i] >= 'A' && value[i] <= 'Z')
                value[i] = static_cast<char>(value[i] + ('a' - 'A'));
        return value;
    }

    // Accepts an HTTP token character for an unquoted parameter or its name.
    bool header_token_char(char c)
    {
        return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
            || (c >= '0' && c <= '9')
            || std::string("!#$%&'*+-.^_`|~").find(c) != std::string::npos;
    }

    // Reads one token or quoted value, including quoted backslash escapes.
    bool read_parameter_value(const std::string &text,
                              std::string::size_type &pos,
                              std::string &value)
    {
        value.clear();
        if (pos == text.size())
            return false;
        if (text[pos] != '"')
        {
            while (pos < text.size() && header_token_char(text[pos]))
                value += text[pos++];
            return !value.empty();
        }
        ++pos;
        while (pos < text.size())
        {
            char c = text[pos++];
            if (c == '"')
                return true;
            if (c == '\\')
            {
                if (pos == text.size())
                    return false;
                c = text[pos++];
            }
            value += c;
        }
        return false;
    }

    // MIME boundaries contain 1-70 allowed ASCII characters, no trailing space.
    bool valid_boundary(const std::string &boundary)
    {
        if (boundary.empty() || boundary.size() > 70
            || boundary[boundary.size() - 1] == ' ')
            return false;
        for (std::string::size_type i = 0; i < boundary.size(); ++i)
        {
            const char c = boundary[i];
            if (!((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')
                || (c >= '0' && c <= '9')
                || std::string("'()+_,-./:=? ").find(c) != std::string::npos))
                return false;
        }
        return true;
    }

    // Extracts a unique boundary; failure leaves boundary unchanged and sets error.
    // Inline allows standalone compilation before parse_multipart uses this helper.
    inline bool extract_boundary(const std::string &content_type,
                                 std::string &boundary,
                                 std::string &error)
    {
        error = "Invalid multipart Content-Type";
        // Reject controls before parsing, including embedded NUL and header injection.
        for (std::string::size_type i = 0; i < content_type.size(); ++i)
        {
            const unsigned char c = static_cast<unsigned char>(content_type[i]);
            if ((c < 32 && c != '\t') || c == 127)
                return false;
        }
        std::string::size_type pos = content_type.find(';');
        if (lower_header_text(trim_header_space(content_type.substr(0, pos)))
            != "multipart/form-data")
            return false;
        bool found = false;
        std::string result;
        while (pos != std::string::npos && pos < content_type.size())
        {
            ++pos;
            pos = content_type.find_first_not_of(" \t", pos);
            if (pos == std::string::npos)
                return false;
            const std::string::size_type start = pos;
            while (pos < content_type.size() && header_token_char(content_type[pos]))
                ++pos;
            const std::string name = lower_header_text(content_type.substr(start, pos - start));
            pos = content_type.find_first_not_of(" \t", pos);
            if (name.empty() || pos == std::string::npos || content_type[pos] != '=')
                return false;
            pos = content_type.find_first_not_of(" \t", pos + 1);
            if (pos == std::string::npos)
                return false;
            std::string value;
            if (!read_parameter_value(content_type, pos, value))
                return false;
            if (name == "boundary")
            {
                if (found || !valid_boundary(value))
                {
                    error = "Invalid or duplicate multipart boundary";
                    return false;
                }
                found = true;
                result = value;
            }
            pos = content_type.find_first_not_of(" \t", pos);
            if (pos != std::string::npos && content_type[pos] != ';')
                return false;
        }
        if (!found)
        {
            error = "Missing multipart boundary";
            return false;
        }
        boundary = result;
        error.clear();
        return true;
    }

    // Checks a complete delimiter line; next points past its CRLF or final EOF.
    bool read_boundary_line(const std::string &body,
                            const std::string &marker,
                            std::string::size_type pos,
                            std::string::size_type &next,
                            bool &closing)
    {
        if (body.compare(pos, marker.size(), marker) != 0)
            return false;
        pos += marker.size();
        closing = body.compare(pos, 2, "--") == 0;
        if (closing)
            pos += 2;
        // MIME delimiter lines may have optional trailing spaces or tabs.
        while (pos < body.size() && (body[pos] == ' ' || body[pos] == '\t'))
            ++pos;
        if (closing && pos == body.size())
        {
            next = pos;
            return true;
        }
        if (body.compare(pos, 2, "\r\n") != 0)
            return false;
        next = pos + 2;
        return true;
    }

    // Finds a full delimiter at a line start, skipping boundary-like file bytes.
    bool find_boundary_line(const std::string &body,
                            const std::string &marker,
                            std::string::size_type start,
                            std::string::size_type &pos,
                            std::string::size_type &next,
                            bool &closing)
    {
        pos = body.find(marker, start);
        while (pos != std::string::npos)
        {
            if ((pos == 0 || (pos >= 2 && body.compare(pos - 2, 2, "\r\n") == 0))
                && read_boundary_line(body, marker, pos, next, closing))
                return true;
            pos = body.find(marker, pos + 1);
        }
        return false;
    }

    // Splits raw parts, preserving headers and binary bytes; no files are written.
    // Preamble and epilogue are ignored. At least one part and a closing line are required.
    // On failure, parts stays unchanged. Inline permits use before public integration.
    inline bool split_multipart_parts(const std::string &body,
                                      const std::string &boundary,
                                      std::vector<std::string> &parts,
                                      std::string &error)
    {
        if (!valid_boundary(boundary))
        {
            error = "Invalid multipart boundary";
            return false;
        }
        const std::string marker = "--" + boundary;
        std::string::size_type pos;
        std::string::size_type next;
        bool closing;
        if (!find_boundary_line(body, marker, 0, pos, next, closing) || closing)
        {
            error = "Missing multipart opening boundary";
            return false;
        }
        std::vector<std::string> result;
        while (!closing)
        {
            const std::string::size_type part_start = next;
            if (!find_boundary_line(body, marker, part_start, pos, next, closing))
            {
                error = "Missing multipart closing boundary";
                return false;
            }
            // Only the CRLF immediately before a delimiter belongs to that delimiter.
            if (pos < 2 || pos - 2 < part_start)
            {
                error = "Missing CRLF before multipart boundary";
                return false;
            }
            result.push_back(body.substr(part_start, pos - 2 - part_start));
        }
        parts.swap(result);
        error.clear();
        return true;
    }

	// Rejects empty names, "." and "..", and any path separators or NUL bytes.
    bool valid_filename(const std::string &filename)
    {
        return !filename.empty()
            && filename != "." && filename != ".."
            && filename.find('/') == std::string::npos
            && filename.find('\\') == std::string::npos
            && filename.find('\0') == std::string::npos;
    }

    // Validates one unfolded part header; control bytes and invalid names are rejected.
    bool parse_part_header(const std::string &line,
                           std::string &name, std::string &value)
    {
        const std::string::size_type colon = line.find(':');
        if (colon == std::string::npos || colon == 0)
            return false;
        for (std::string::size_type i = 0; i < colon; ++i)
            if (!header_token_char(line[i]))
                return false;
        for (std::string::size_type i = colon + 1; i < line.size(); ++i)
        {
            const unsigned char c = static_cast<unsigned char>(line[i]);
            if ((c < 32 && c != '\t') || c == 127)
                return false;
        }
        name = lower_header_text(line.substr(0, colon));
        value = trim_header_space(line.substr(colon + 1));
        return true;
    }

    // Reads unique form-data parameters, reusing the quoted-value parser.
    bool parse_disposition(const std::string &text,
                           std::string &filename, bool &is_file)
    {
        std::string::size_type pos = text.find(';');
        if (lower_header_text(trim_header_space(text.substr(0, pos))) != "form-data")
            return false;
        std::map<std::string, std::string> parameters;
        while (pos != std::string::npos && pos < text.size())
        {
            pos = text.find_first_not_of(" \t", pos + 1);
            if (pos == std::string::npos)
                return false;
            const std::string::size_type start = pos;
            while (pos < text.size() && header_token_char(text[pos]))
                ++pos;
            const std::string name = lower_header_text(text.substr(start, pos - start));
            pos = text.find_first_not_of(" \t", pos);
            if (name.empty() || pos == std::string::npos || text[pos] != '=')
                return false;
            pos = text.find_first_not_of(" \t", pos + 1);
            if (pos == std::string::npos)
                return false;
            std::string value;
            if (!read_parameter_value(text, pos, value)
                || parameters.find(name) != parameters.end())
                return false;
            parameters[name] = value;
            pos = text.find_first_not_of(" \t", pos);
            if (pos != std::string::npos && text[pos] != ';')
                return false;
        }
        // Extended filename encoding is unsupported; do not silently misname a file.
        if (parameters.find("name") == parameters.end() || parameters["name"].empty()
            || parameters.find("filename*") != parameters.end())
            return false;
        is_file = parameters.find("filename") != parameters.end();
        if (is_file)
        {
            filename = parameters["filename"];
            return valid_filename(filename);
        }
        return true;
    }

    // Requires one Content-Disposition; duplicate headers and transfer encodings fail.
    bool read_part_headers(const std::string &headers,
                           std::string &filename, bool &is_file)
    {
        std::map<std::string, std::string> values;
        std::string::size_type pos = 0;
        while (pos < headers.size())
        {
            const std::string::size_type end = headers.find("\r\n", pos);
            const std::string line = headers.substr(pos,
                end == std::string::npos ? std::string::npos : end - pos);
            std::string name;
            std::string value;
            if (!parse_part_header(line, name, value)
                || values.find(name) != values.end())
                return false;
            values[name] = value;
            if (end == std::string::npos)
                break;
            pos = end + 2;
        }
        // Only raw part bytes are supported; no base64 or nested multipart decoding.
        if (values.find("content-disposition") == values.end()
            || values.find("content-transfer-encoding") != values.end())
            return false;
        if (values.find("content-type") != values.end()
            && lower_header_text(values["content-type"]).compare(0, 10, "multipart/") == 0)
            return false;
        return parse_disposition(values["content-disposition"], filename, is_file);
    }

    // Separates headers from bytes; normal form fields are validated but not returned.
    bool parse_file_part(const std::string &part, Upload::UploadedFile &file,
                         bool &is_file, std::string &error)
    {
        const std::string::size_type header_end = part.find("\r\n\r\n");
        if (header_end == std::string::npos)
        {
            error = "Missing multipart header separator";
            return false;
        }
        if (!read_part_headers(part.substr(0, header_end), file.filename, is_file))
        {
            error = "Invalid multipart headers or filename";
            return false;
        }
        // substr uses a byte count, so NUL bytes and trailing CRLF remain intact.
        if (is_file)
            file.content = part.substr(header_end + 4);
        return true;
    }

	// Checks that the directory exists, is a directory, and is writable and searchable.
    bool valid_directory(const std::string &directory)
    {
        if (directory.empty() || directory.find('\0') != std::string::npos)
        {
            errno = EINVAL;
            return false;
        }
        struct stat info;
        if (stat(directory.c_str(), &info) == -1)
            return false;
        if (!S_ISDIR(info.st_mode))
        {
            errno = ENOTDIR;
            return false;
        }
        return access(directory.c_str(), W_OK | X_OK) == 0;
    }

	// Writes the entire content to the file descriptor, handling partial writes and EINTR.
    bool write_content(int fd, const std::string &content)
    {
        std::string::size_type offset = 0;
        const std::size_t max_write = static_cast<std::size_t>(
            std::numeric_limits<ssize_t>::max());

        while (offset < content.size())
        {
            std::size_t count = content.size() - offset;
            if (count > max_write)
                count = max_write;
            ssize_t written = write(fd, content.c_str() + offset, count);
            if (written < 0)
            {
                if (errno == EINTR)
                    continue;
                return false;
            }
            if (written == 0)
            {
                errno = EIO;
                return false;
            }
            offset += static_cast<std::string::size_type>(written);
        }
        return true;
    }
}

// Saves a file to the specified directory, rejecting unsafe filenames and existing files.
// On failure, returns false and sets errno. Existing files are preserved.
bool Upload::save_file(const std::string &directory,
                       const std::string &filename,
                       const std::string &content)
{
    if (!valid_filename(filename))
    {
        errno = EINVAL;
        return false;
    }
    if (!valid_directory(directory))
        return false;

    std::string path = directory;
    if (path[path.size() - 1] != '/')
        path += '/';
    path += filename;

    int fd = open(path.c_str(), O_WRONLY | O_CREAT | O_EXCL, 0600);
    if (fd == -1)
        return false;
    if (!write_content(fd, content))
    {
        const int error = errno;
        close(fd);
        unlink(path.c_str());
        errno = error;
        return false;
    }
    if (close(fd) == -1)
    {
        const int error = errno;
        unlink(path.c_str());
        errno = error;
        return false;
    }
    return true;
}


// Parses the complete collected body; it performs no filesystem or socket operations.
// Results replace files only after every part passes validation, including form fields.
bool Upload::parse_multipart(const std::string &content_type,
                             const std::string &body,
                             std::vector<UploadedFile> &files,
                             std::string &error)
{
    std::string boundary;
    std::vector<std::string> parts;
    if (!extract_boundary(content_type, boundary, error)
        || !split_multipart_parts(body, boundary, parts, error))
        return false;
    std::vector<UploadedFile> result;
    for (std::size_t i = 0; i < parts.size(); ++i)
    {
        UploadedFile file;
        bool is_file = false;
        if (!parse_file_part(parts[i], file, is_file, error))
            return false;
        if (is_file)
            result.push_back(file);
    }
    files.swap(result);
    error.clear();
    return true;
}

webserv::webserv()

webserv::webserv(const std::string &path)
{
    this->_config_path = path;

webserv::webserv(const webserv &ot)
{
    *this = ot;
}

webserv& webserv::operator=(const webserv &ot)
{
    if(this != &ot)
        this->_config_path = ot._config_path;
    return(*this);
}

void webserv::check_path(std::string &path)
{
    struct stat info;

    if (stat(path.c_str(), &info) != 0)
        throw std::runtime_error(
            "Cannot access configuration file: " + path);

    if (!S_ISREG(info.st_mode))
        throw std::runtime_error(
            "Configuration path is not a regular file: " + path);

    if (access(path.c_str(), R_OK) != 0)
        throw std::runtime_error(
            "Configuration file is not readable: " + path);
}


void webserv::take_path()
{
    if (this->_config_path == "")
        this->_config_path = "config/default.conf";

    this->check_path(this->_config_path);
}



std::string webserv::get_path()
{
    return(this->_config_path);
}

{
